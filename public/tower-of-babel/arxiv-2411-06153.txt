\mathscr{D}_{\nu}=\mathscr{D}(i_{1},\ldots,i_{\nu})=\{1,2,\ldots,r\}\setminus\{i_{1},\ldots,i_{\nu}\}

\displaystyle\prod_{k=1}^{r}

\displaystyle\left(T(\alpha;N_{k},H_{k})+\theta_{k}\right)=\prod_{k=1}^{r}T(\alpha;N_{k},H_{k})+

\displaystyle+\sum_{\nu=1}^{r-1}\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\prod_{j=1}^{\nu}T(\alpha;N_{i_{j}},H_{i_{j}})\prod_{k\in\mathscr{D}_{\nu}}\theta_{k}+\prod_{k=1}^{r}\theta_{k},

we represent

In the sum

, going to estimates and using the fact that the geometric mean of non-negative numbers does not exceed their arithmetic mean, and then associating each
\nu

\displaystyle R_{n,r}(N,H)

\displaystyle\leq\sum_{\nu=1}^{r-1}\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\int_{0}^{1}\prod_{j=1}^{\nu}|T(\alpha;N_{i_{j}},H_{i_{j}})|d\alpha\leq

\displaystyle\leq\sum_{\nu=1}^{r-1}\frac{I(\nu)}{\nu}=I(1)+\sum_{k=1}^{n-1}\sum_{\nu=2^{k}}^{2^{k+1}-1}\frac{I(\nu)}{\nu}+\frac{I(r-1)}{r-1},

\displaystyle I(\nu)=\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\sum_{j=1}^{\nu}\int_{0}^{1}\left|T(\alpha;N_{i_{j}},H_{i_{j}})\right|^{\nu}d\alpha.

For

, let’s estimate

from above. Using the trivial estimate of the sum
|T(\alpha;N_{i_{j}},H_{i_{j}})|
and Theorem

\displaystyle I(\nu)

\displaystyle\leq\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\sum_{j=1}^{\nu}\max\left|T(\alpha;N_{i_{j}},H_{i_{j}})\right|^{\nu-2^{k}}\int_{0}^{1}\left|T(\alpha;N_{i_{j}},H_{i_{j}})\right|^{2^{k}}d\alpha\leq

5.1.

By the definition of the integral

\displaystyle J(\mathfrak{M}_{1})=\hskip-14.0pt\sum_{q\leq H_{r}\mathscr{L}^{-1}}\sum_{\begin{subarray}{c}a=0\\

For the sum

) and (

\begin{split}\tau=&2(n-1)nN_{1}^{n-2}H_{1}=\frac{4(n-1)H}{\mu_{1}^{\frac{1}{n}}N^{\frac{1}{n}}}\left(1+O\left(\frac{H}{N}\right)\right)\geq\\
&\geq\frac{4(n-1)H}{\mu_{k}^{\frac{1}{n}}N^{\frac{1}{n}}}\left(1+O\left(\frac{H}{N}\right)\right)=2(n-1)nN_{k}^{n-2}H_{k},\end{split}

and from the relations

,

So, according to this corollary, for

Multiplying both sides of these formulas for all

\displaystyle\prod_{k=1}^{r}T\left(\frac{a}{q}+\lambda,N_{k},H_{k}\right)

\displaystyle\prod_{k=1}^{r}T

\displaystyle\left(\frac{a}{q}+\lambda,N_{k},H_{k}\right)-\frac{S^{r}(a,q)}{q^{r}}\prod_{k=1}^{r}H_{k}\gamma(\lambda;N_{k},H_{k})\ll

\displaystyle\ll\sum_{\nu=1}^{r-1}q^{\left(\frac{1}{2}+\varepsilon\right)(r-\nu)-\frac{\nu}{n}}\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\prod_{k=1}^{\nu}H_{i_{k}}|\gamma(\lambda;N_{i_{k}},H_{i_{k}})|+q^{0,5r+r\varepsilon}.

From here and from the formula (

\displaystyle J(\mathfrak{M}_{1})

(a,q)=1\end{subarray}}^{q-1}\hskip-8.0pt\frac{S^{r}(a,q)}{q^{r}}e\left(-\frac{aN}{q}\right)+R_{1}(\mathfrak{M}_{1})+R_{2}(\mathfrak{M}_{1}),

5.4.
Calculation of the integral

We have

\displaystyle\prod_{k=1}^{r}(a_{k}+b)=\prod_{k=1}^{r}a_{k}+b^{r}+\sum_{\nu=1}^{r-1}b^{r-\nu}\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\prod_{k=1}^{\nu}a_{i_{k}},

\displaystyle\prod_{k=1}^{r}\gamma(\lambda;N_{k},H_{k})

\displaystyle R_{5}(N,

\displaystyle H)=R_{4}^{r}(N,H)+\sum_{\nu=1}^{r-1}R_{4}^{r-\nu}(N,H)\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}\prod_{k=1}^{\nu}e\left(\mu_{i_{k}}N\lambda\right)\frac{\sin\left(2\pi H\lambda\right)}{2\pi H\lambda}=

\displaystyle=R_{4}^{r}(N,H)+\sum_{\nu=1}^{r-1}\frac{\sin^{\nu}\left(2\pi H\lambda\right)}{(2\pi H\lambda)^{\nu}}R_{4}^{r-\nu}(N,H)\sum_{1\leq i_{1}<\ldots<i_{\nu}\leq r}e\left(\lambda N\sum_{k=1}^{\nu}\mu_{i_{k}}\right)\ll

\displaystyle\ll\sum_{\nu=0}^{r-1}\frac{|\sin(2\pi H\lambda)|^{\nu}}{|2\pi\lambda H|^{\nu}}\left(\frac{H\mathscr{L}}{N}\right)^{r-\nu}\ll\frac{H^{r}\mathscr{L}^{r}}{N^{r}}+\frac{|\sin(2\pi H\lambda)|^{r-1}}{|2\pi\lambda H|^{r-1}}\cdot\frac{H\mathscr{L}}{N}.

From here, using the definition of the integral

and (

\displaystyle\mathscr{A}_{1}(r,q)

\displaystyle=\int_{|\lambda|\leq\eta}\left(\frac{\sin^{r}\left(2\pi H\lambda\right)}{(2\pi H\lambda)^{r}}e\left(N\lambda\right)+R_{5}(N,H)\right)e(-\lambda N)d\lambda=

\displaystyle=\frac{1}{\pi H}\int_{0}^{2\pi H\eta}\frac{\sin^{r}t}{t^{r}}dt+R_{6}(N,H)+R_{7}(N,H),

\displaystyle R_{6}(N,H)

\displaystyle\ll\int_{|\lambda|\leq\eta}\frac{H^{r}\mathscr{L}^{r}}{N^{r}}d\lambda\leq\frac{2H^{r-1}\mathscr{L}^{r+1}}{N^{r}}=\frac{1}{H\mathscr{L}^{7}}\cdot\frac{H^{r}\mathscr{L}^{r+8}}{N^{r}}\ll\frac{1}{H\mathscr{L}^{7}},

\displaystyle R_{8}(N,H)=\frac{1}{\pi H}\int_{2\pi H\eta}^{\infty}\frac{\sin^{r}t}{t^{r}}dt\leq\frac{1}{\pi H}\cdot\frac{1}{(2\pi H\eta)^{r}}\ll\frac{1}{H^{r+1}\eta^{r}}=

\displaystyle=\frac{1}{H^{r+1}}\cdot\left(\frac{2nN_{r}^{n-1}H_{r}}{\mathscr{L}}\right)^{r}\ll\frac{1}{H^{r+1}\mathscr{L}^{r}}\left(\frac{2H}{n}\right)^{r}\ll\frac{1}{H\mathscr{L}^{r}}.

Using the lemma

to calculate the improper integral with

\displaystyle\mathscr{A}_{1}(r,q)=\frac{\gamma(n,r)}{H}+O\left(\frac{1}{H\mathscr{L}^{7}}\right).

\displaystyle\gamma(n,r)=

\displaystyle\frac{r^{r-1}-\frac{r}{1!}(r-2)^{r-1}+\frac{r(r-1)}{2!}(r-4)^{r-1}-\frac{r(r-1)(r-2)}{3!}(r-6)^{r-1}+\ldots}{2^{r}(r-1)!}.

Now let’s estimate the integral

\displaystyle=\frac{2}{H^{r}}\int_{\eta}^{\eta_{q}}\frac{d\lambda}{\lambda^{r}}=\frac{2}{(r-1)H^{r}}\left(\frac{1}{\eta^{r-1}}-\frac{1}{\eta_{q}^{r-1}}\right)\leq\frac{2}{(r-1)H^{r}\eta^{r-1}}=

\displaystyle\mathscr{A}(r,q)=\frac{\gamma(n,r)}{H}+O\left(\frac{1}{H\mathscr{L}^{r-1}}\right).

5.6.
Estimation of the integral

We have

\displaystyle J(\mathfrak{M}_{2})=\int_{\mathfrak{M}_{2}}\prod_{k=1}^{r}T(\alpha;N_{k},H_{k})e(-\alpha N)d\alpha.

The sums

60

1.3
and the relation
H_{\nu}\leq H_{1}\ll HN^{-1+\frac{1}{n}}

\displaystyle J(\mathfrak{M}_{2})

\displaystyle\leq\max_{\alpha\in\mathfrak{M}_{2}}\left|T(\alpha;N_{\nu},H_{\nu})\right|\int_{0}^{1}\prod_{\begin{subarray}{c}k=1\\
k\neq\nu\end{subarray}}^{r}\left|T(\alpha;N_{k},H_{k})\right|d\alpha\leq

k\neq\nu\end{subarray}}^{r}\int_{0}^{1}\left|T(\alpha;N_{k},H_{k})\right|^{r-1}d\alpha\ll
