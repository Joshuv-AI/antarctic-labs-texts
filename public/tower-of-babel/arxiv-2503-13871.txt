\begin{split}P_{1}=&-A_{0}\partial_{t}\bm{\phi}-R_{i}R_{k}A^{k}\partial^{i}\bm{\phi}\\

\begin{split}|\bm{\phi}|_{s,b}+|\mathbf{A}|_{s-\frac{1}{2},b}\leq 2(1+C_{0})M_{0}.\end{split}

Combining the local existence and uniform bounds in Sect.
3.1
, we have proved Theorem
1.1
.

4.
appendix: Energy conservation

We present the detailed calculation of the energy conservation law as given in equation (
1.3
):

E(\bm{\phi},A)=\frac{1}{2}\int_{\mathbb{R}^{2}}\left(|D_{\mu}\bm{\phi}|^{2}+\frac{1}{\kappa^{2}}(1+n_{3}\cdot\bm{\phi})(1-n_{3}\cdot\bm{\phi})^{3}\right)dx.

We first take the inner product of both sides of
\eqref{E-L}_{1}
with
D_{0}\bm{\phi}
to get

\displaystyle 0=

\displaystyle D_{0}\bm{\phi}\cdot D_{0}D_{0}\bm{\phi}-D_{0}\bm{\phi}\cdot D_{1}D_{1}\bm{\phi}-D_{0}\bm{\phi}\cdot D_{2}D_{2}\bm{\phi}+(D_{\mu}\bm{\phi}\cdot D_{\mu}\bm{\phi})(\bm{\phi}\cdot D_{\mu}\bm{\phi})

\displaystyle+\frac{1}{\kappa^{2}}(1-n_{3}\cdot\bm{\phi})^{2}(1+2n_{3}\cdot\bm{\phi})\Big[(n_{3}\cdot\bm{\phi})(\bm{\phi}\cdot D_{0}\bm{\phi})-n_{3}\cdot D_{0}\bm{\phi}\big]

\displaystyle=

\displaystyle\frac{1}{2}\partial_{0}|D_{0}\bm{\phi}|^{2}-\partial_{1}(D_{0}\bm{\phi}\cdot D_{1}\bm{\phi})+D_{1}D_{0}\bm{\phi}\cdot D_{1}\bm{\phi}-\partial_{2}(D_{0}\bm{\phi}\cdot D_{2}\bm{\phi})+D_{2}D_{0}\bm{\phi}\cdot D_{2}\bm{\phi}

\displaystyle+|D_{\mu}\bm{\phi}|^{2}\partial_{0}\frac{|\bm{\phi}|^{2}}{2}+\frac{1}{\kappa^{2}}(1-n_{3}\cdot\bm{\phi})^{2}(1+2n_{3}\cdot\bm{\phi})\left[\left((n_{3}\cdot\bm{\phi})\partial_{0}\frac{|\bm{\phi}|^{2}}{2}\right)-\partial_{0}(n_{3}\cdot\bm{\phi})\right]

\displaystyle=

\displaystyle\frac{1}{2}\partial_{0}|D_{0}\bm{\phi}|^{2}-\partial_{1}(D_{0}\bm{\phi}\cdot D_{1}\bm{\phi})+D_{1}D_{0}\bm{\phi}\cdot D_{1}\bm{\phi}-\partial_{2}(D_{0}\bm{\phi}\cdot D_{2}\bm{\phi})+D_{2}D_{0}\bm{\phi}\cdot D_{2}\bm{\phi}

\displaystyle-\frac{1}{\kappa^{2}}(1-n_{3}\cdot\bm{\phi})^{2}(1+2n_{3}\cdot\bm{\phi})\partial_{0}(n_{3}\cdot\bm{\phi}),

where we use the constraint condition
|\bm{\phi}|=1
. Next, we integrate the above equality over
\mathbb{R}^{2}
to obtain:

\displaystyle 0=

\displaystyle\int_{\mathbb{R}^{2}}\frac{1}{2}\partial_{0}|D_{0}\bm{\phi}|^{2}+D_{1}D_{0}\bm{\phi}\cdot D_{1}\bm{\phi}+D_{2}D_{0}\bm{\phi}\cdot D_{2}\bm{\phi}-\frac{1}{\kappa^{2}}(1-n\cdot\bm{\phi})^{2}(1+2n_{3}\cdot\bm{\phi})\partial_{0}(n_{3}\cdot\bm{\phi})dx

\displaystyle=

\displaystyle\int_{\mathbb{R}^{2}}\frac{1}{2}\partial_{0}|D_{0}\bm{\phi}|^{2}+(D_{0}D_{1}\bm{\phi}+F_{10}n_{3}\times\bm{\phi})\cdot D_{1}\bm{\phi}+(D_{0}D_{2}\bm{\phi}+F_{20}n_{3}\times\bm{\phi})\cdot D_{2}\bm{\phi}

\displaystyle+\frac{1}{2}\partial_{0}\left(\frac{1}{\kappa^{2}}(1+n_{3}\cdot\bm{\phi})(1-n_{3}\cdot\bm{\phi})^{3}\right)dx

\displaystyle=

\displaystyle\int_{\mathbb{R}^{2}}\frac{1}{2}\partial_{0}(|D_{0}\bm{\phi}|^{2}+|D_{1}\bm{\phi}|^{2}+|D_{2}\bm{\phi}|^{2})+F_{10}n\times\bm{\phi}\cdot D_{1}\bm{\phi}+F_{20}n_{3}\times\bm{\phi}\cdot D_{1}\bm{\phi}

\displaystyle+\frac{1}{2}\partial_{0}\left(\frac{1}{\kappa^{2}}(1+n_{3}\cdot\bm{\phi})(1-n_{3}\cdot\bm{\phi})^{3}\right)dx

\displaystyle=

\displaystyle\int_{\mathbb{R}^{2}}\frac{1}{2}\partial_{0}(|D_{0}\bm{\phi}|^{2}+|D_{1}\bm{\phi}|^{2}+|D_{2}\bm{\phi}|^{2})-\frac{1}{\kappa}(n_{3}\cdot\bm{\phi}\times D_{2}\bm{\phi})(n_{3}\times\bm{\phi}\cdot D_{1}\bm{\phi})

\displaystyle+\frac{1}{\kappa}(n_{3}\cdot\bm{\phi}\times D_{1}\bm{\phi})(n_{3}\times\bm{\phi}\cdot D_{2}\bm{\phi})+\frac{1}{2}\partial_{0}\left(\frac{1}{\kappa^{2}}(1+n_{3}\cdot\bm{\phi})(1-n_{3}\cdot\bm{\phi})^{3}\right)dx

\displaystyle=

\displaystyle\int_{\mathbb{R}^{2}}\frac{1}{2}\partial_{0}(|D_{0}\bm{\phi}|^{2}+|D_{1}\bm{\phi}|^{2}+|D_{2}\bm{\phi}|^{2})+\frac{1}{2}\partial_{0}\left(\frac{1}{\kappa^{2}}(1+n_{3}\cdot\bm{\phi})(1-n\cdot\bm{\phi})^{3}\right)dx.

This yields the conserved energy equation as desired.
