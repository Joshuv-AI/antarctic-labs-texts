Abstract
. For 3-D quadratic quasilinear wave equations with or without null conditions in exterior domains, when the compatible initial data and Dirichlet boundary values are given, the global existence or the maximal existence time of small data smooth solutions have been established in early references. For the Cauchy problem of 2-D quadratic quasilinear wave equations with null conditions, it has been shown that the small data smooth solutions exist globally. However, for the corresponding 2-D initial boundary value problem in exterior domains, it is still open whether the global solutions exist. In the present paper, we solve this open problem through proving the global existence of small solutions in exterior domains. Our main ingredients include: deriving new precise pointwise estimates for the initial boundary value problem of 2-D linear wave equations in exterior domains; finding appropriate divergence structures of quasilinear wave equations under null conditions; introducing a good unknown to eliminate the resulting Q 0 Q_{0} type nonlinearity, and establishing some crucial pointwise spacetime decay estimates of solutions and their derivatives. Keywords. Quadratic quasilinear wave equation, initial boundary value problem, null condition, divergence structure, good unknown, global smooth solution 2020 Mathematical Subject Classification. 35L05, 35L20, 35L70

\begin{split}Q_{0i}(\partial_{j}f,g)&=\omega_{i}\omega_{j}\Box f\partial_{t}g-\omega_{i}\omega_{j}Q_{0}(\partial_{t}f,g)-\sum_{k=1,2}\partial_{t}g\frac{\omega_{k}\Omega_{ki}\partial_{j}f+\omega_{i}\Omega_{kj}\partial_{k}f}{|x|}\\
&\quad+\sum_{k=1,2}\frac{\omega_{k}\partial^{2}_{0j}f\Omega_{ki}g+\omega_{i}\Omega_{kj}\partial_{t}f\partial_{k}g}{|x|}.\end{split}

This, together with (

At last, we rewrite the last two summations of (

Due to

\begin{split}\sum_{k=1,2}[\omega_{k},\Omega_{ki}]F&=\sum_{k=1,2}(-x_{k}\delta_{ik}+x_{i})\frac{F}{|x|}=\omega_{i}F,\\
\sum_{k=1,2}[\Omega_{kj},\partial_{k}]F&=\sum_{k=1,2}(-\partial_{j}+\delta_{jk}\partial_{k})F=-\partial_{j}F,\end{split}

then one has

(2.13)

\begin{split}&-\sum_{k=1,2}\partial_{t}g\frac{\omega_{k}\Omega_{ki}\partial_{j}f+\omega_{i}\Omega_{kj}\partial_{k}f}{|x|}+\sum_{k=1,2}\frac{\omega_{k}\partial^{2}_{0j}f\Omega_{ki}g+\omega_{i}\Omega_{kj}\partial_{t}f\partial_{k}g}{|x|}\\

&\quad+\sum_{k=1,2}\frac{\omega_{i}}{|x|}(\Omega_{kj}\partial_{t}f\partial_{k}g+\Omega_{kj}\partial^{2}_{0k}fg)\\
&=-\sum_{k=1,2}\partial_{t}(g\frac{\omega_{k}\Omega_{ki}\partial_{j}f+\omega_{i}\Omega_{kj}\partial_{k}f}{|x|})+\sum_{k=1,2}\Omega_{ki}(\frac{\omega_{k}\partial^{2}_{0j}fg}{|x|})+\frac{\omega_{i}\partial^{2}_{0j}fg}{|x|}\\
&\quad+\sum_{k=1,2}\partial_{k}(\frac{\omega_{i}\Omega_{kj}\partial_{t}fg}{|x|})-\frac{\omega_{i}\partial^{2}_{0j}fg}{|x|}-\sum_{k=1,2}\partial_{k}(\frac{\omega_{i}}{|x|})\Omega_{kj}\partial_{t}fg.\end{split}

Therefore, by (
2.12

), we complete the proof of Lemma

.

K^{\pm}_{k}(t,x)=\frac{2^{k}}{(2\pi)^{2}}\int\int_{\mathop{\mathbb{S}\kern 0.0pt}\nolimits}e^{i2^{k}\rho(\pm t+x\cdot w)}\frac{\psi_{0}(\rho)}{(1+2^{2k}\rho^{2})}dwd\rho,

where we have used the fact of
\psi_{0}(x)=\psi_{0}(|x|)
.
Recalling Corollary 2.37 of
[
36
]
, one has

(4.9)

\begin{split}&\widehat{\sigma_{\mathop{\mathbb{S}\kern 0.0pt}\nolimits}}(x)=\int_{\mathop{\mathbb{S}\kern 0.0pt}\nolimits}e^{ix\cdot w}dw=e^{i|x|}\omega_{+}(|x|)+e^{-i|x|}\omega_{-}(|x|),\\
&|\partial_{r}^{l}\omega_{\pm}(r)|\lesssim r^{-1/2-l}\qquad\text{for $r\geq 1$ and $l\geq 0$}.\end{split}

Therefore, (
4.8
) is reduced to

(4.10)

\begin{split}K^{+}_{k}(t,x)&=\frac{2^{k}}{(2\pi)^{2}}\int(e^{i2^{k}\rho(t+|x|)}\omega_{+}(2^{k}\rho|x|)+e^{i2^{k}\rho(t-|x|)}\omega_{-}(2^{k}\rho|x|))\frac{\psi_{0}(\rho)}{(1+2^{2k}\rho^{2})}d\rho\\
&:=K^{++}_{k}(t,x)+K^{+-}_{k}(t,x).\end{split}

It follows from (
4.9
) and (
4.10
) with
2^{k}|x|\geq 10
that

(4.11)

|K^{+\pm}_{k}(t,x)|\lesssim 2^{k-2k_{+}}(2^{k}|x|)^{-1/2}\lesssim 2^{k-2k_{+}}(1+2^{k}(t+|x|))^{-1/2},

which derives (
4.3
) in the region
2^{k}|t-|x||\leq 1
.

For
2^{k}|t-|x||\geq 1
, through integrating by parts with respect to
\rho
in
K^{+-}_{k}(t,x)
we arrive at

(4.12)

\begin{split}|K^{+-}_{k}(t,x)|&=\Big|\frac{2^{k}}{(2\pi)^{2}}\int\frac{\partial_{\rho}e^{i2^{k}\rho(t-|x|)}}{i2^{k}(t-|x|)}\omega_{-}(2^{k}\rho|x|)\frac{\psi_{0}(\rho)}{(1+2^{2k}\rho^{2})}d\rho\Big|\\
&\lesssim 2^{k}(2^{k}(|t-|x||)^{-1}\int\Big|\partial_{\rho}\Big(\frac{\omega_{-}(2^{k}\rho|x|)\psi_{0}(\rho)}{(1+2^{2k}\rho^{2})}\Big)\Big|d\rho\\
&\lesssim 2^{k-2k_{+}}(2^{k}|t-|x||)^{-1}(2^{k}|x|)^{-1/2}\\
&\lesssim 2^{k-2k_{+}}(1+2^{k}|t-|x||)^{-1}(1+2^{k}(t+|x|))^{-1/2}.\\
\end{split}

The estimates of
K^{++}_{k}(t,x)
and
K^{-}_{k}(t,x)
are analogous.
Thus, collecting (
4.5
), (
4.7
) and (
4.10
)-(
4.12
) yields (
4.3
).

Based on (
4.1
) and (
4.3

3.8

For fixed

(4.13)

2^{\bar{K}}\approx\langle{t+|x|}\rangle^{-2},\qquad|\bar{K}|\lesssim\ln(2+t+|x|).

Thus, (

) can be separated into such three parts

(4.14)

\begin{split}v_{low}(t,x)&:=\int_{0}^{t}\int_{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}\sum_{k<\bar{K}}(K^{+}_{k}-K^{-}_{k})(t-s,x-y)(1-\Delta)H(s,y)dyds,\\
v_{-}(t,x)&:=\int_{0}^{t}\int_{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}\sum_{\bar{K}\leq k<0}(K^{+}_{k}-K^{-}_{k})(t-s,x-y)(1-\Delta)H(s,y)dyds,\\
v_{+}(t,x)&:=\int_{0}^{t}\int_{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}\sum_{k\geq 0}(K^{+}_{k}-K^{-}_{k})(t-s,x-y)(1-\Delta)H(s,y)dyds.\end{split}

Due to

4.5

|v_{low}|\lesssim 2^{\bar{K}}t\sup_{(s,y)\in[0,t]\times\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}|(1-\Delta)H(s,y)|\lesssim\langle{t+|x}\rangle^{-1}\sup_{(s,y)\in[0,t]\times\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}|(1-\Delta)H(s,y)|.

On the other hand, one has

\frac{1}{1+2^{k}(t+|x|)}\lesssim\left\{\begin{aligned} &\frac{1}{2^{k}(1+t+|x|)},&&\qquad k\leq 0,\\
&\frac{1}{1+t+|x|},&&\qquad k\geq 0,\end{aligned}\right.

which leads to

(1+2^{k}|t+|x||)^{-1/2}(1+2^{k}|t-|x||)^{-1/2}\lesssim 2^{k_{+}-k}(1+t+|x|)^{-1/2}(1+|t-|x||)^{-1/2}.

Together with (

\begin{split}|v_{-}|&\lesssim I^{\pm}|\bar{K}|\sup_{(s,y)\in[0,t]\times\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}(1+s)^{1/2+\mu}|(1-\Delta)H(s,y)|\\

|v_{+}|&\lesssim I^{\pm}\sup_{(s,y)\in[0,t]\times\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}(1+s)^{1/2+\mu}|(1-\Delta)H(s,y)|,\end{split}

where

I^{\pm}:=\int_{|y|\leq R}\int_{0}^{t}\frac{dsdy}{(1+t-s+|x-y|)^{1/2}(1+|t-s-|x-y||)^{1/2}(1+s)^{1/2+\mu}}.

We can assert

(4.17)

I^{\pm}\lesssim\frac{\ln(2+t+|x|)}{\langle{t+|x|}\rangle^{1/2}\langle{t-|x|}\rangle^{\mu}}.

Note that if (
4.17
) is proved, then Lemma
3.8

4.14
), (
4.15
)
and (
4.16
).
In the rest part, we are devoted to the proof of assertion (

Set

with

(4.18)

\begin{split}I^{\pm}_{1}&:=\int_{|y|\leq R,|x-y|\geq t}\int_{0}^{t}\frac{(1+s)^{-1/2-\mu}dsdy}{(1+t-s+|x-y|)^{1/2}(1+|t-s-|x-y||)^{1/2}},\\
I^{\pm}_{2}&:=\int_{|y|\leq R,|x-y|\leq t}\int_{0}^{t}\frac{(1+s)^{-1/2-\mu}dsdy}{(1+t-s+|x-y|)^{1/2}(1+|t-s-|x-y||)^{1/2}}.\end{split}

.

Next, we estimate

, where

(4.20)

\begin{split}I^{\pm}_{21}&:=\int_{\begin{subarray}{c}|y|\leq R,\\

&\lesssim 2^{5(k_{+}-k)/4}\langle{t-|x|}\rangle^{-1}\int_{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}\frac{dy}{(1+2^{k-k_{+}}|y|)^{3}}\\
&\lesssim 2^{13(k_{+}-k)/4}\langle{t-|x|}\rangle^{-1}.\end{split}

For

Collecting (

\begin{split}|\partial_{j}v|&\lesssim\langle{x}\rangle^{-1/2}\langle{t-|x|}\rangle^{-1}\sup_{(s,y)\in[0,t]\times\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}}(1+s)|(1-\Delta)^{3}H(s,y)|\sum_{k\in\mathop{\mathbb{Z}\kern 0.0pt}\nolimits}2^{k/4-9k_{+}/4}\\

5.
Pointwise estimates of the initial boundary value problem

Lemma 5.1
.

(
Pointwise estimates in exterior domain
)
Suppose that the obstacle
\mathcal{O}
is star-shaped,
\mathcal{K}=\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}\setminus\mathcal{O}
and
w(t,x)
is the solution of the IBVP

\left\{\begin{aligned} &\Box w=F(t,x),\qquad(t,x)\in[0,\infty)\times\mathcal{K},\\
&w|_{\partial\mathcal{K}}=0,\\
&(w,\partial_{t}w)(0,x)=(w_{0},w_{1})(x),\quad x\in\mathcal{K},\end{aligned}\right.

where
(w_{0},w_{1})
admits a compact support and
\operatorname{supp}_{x}F(t,x)\subset\{x:|x|\leq t+M_{0}\}
.
Then we have that for any
\mu,\nu\in(0,1/2)
,

(5.1)

\begin{split}&\langle{x}\rangle^{1/2}\langle{t-|x|}\rangle^{1/2-\mu}|\partial w|\lesssim\|(w_{0},w_{1})\|_{H^{4}(\mathcal{K})}+\sum_{|a|\leq 3}\sup_{s\in[0,t]}\langle{s}\rangle^{1/2}\|\partial^{a}F(s)\|_{L^{2}(\mathcal{K}_{4})}\\
&+\sum_{|a|\leq 3}\sup_{y\in\mathcal{K}}|\partial^{a}F(0,y)|+\sum_{|a|\leq 4}\sup_{(s,y)\in[0,t]\times(\overline{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}\setminus\mathcal{K}_{2}})}\langle{y}\rangle^{1/2}\mathcal{W}_{1+\mu+\nu,1}(s,y)|Z^{a}F(s,y)|\end{split}

and

(5.2)

\begin{split}&\frac{\langle{t+|x|}\rangle^{1/2}\langle{t-|x|}\rangle^{\mu}}{\ln^{2}(2+t+|x|)}|w|\lesssim\|(w_{0},w_{1})\|_{H^{5}(\mathcal{K})}+\sum_{|a|\leq 4}\sup_{s\in[0,t]}\langle{s}\rangle^{1/2+\mu}\|\partial^{a}F(s)\|_{L^{2}(\mathcal{K}_{4})}\\
&+\sum_{|a|\leq 4}\sup_{y\in\mathcal{K}}|\partial^{a}F(0,y)|+\sum_{|a|\leq 5}\sup_{(s,y)\in[0,t]\times(\overline{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}\setminus\mathcal{K}_{2}})}\langle{y}\rangle^{1/2}\mathcal{W}_{1+\mu,1+\nu}(s,y)|\partial^{a}F(s,y)|.\end{split}

In addition, let

solve

(5.3)

\left\{\begin{aligned} &\Box w^{div}=F(t,x)=G+\partial_{\theta}G^{\theta}+\sum_{\alpha=0}^{2}\partial_{\alpha}G^{\alpha},\qquad(t,x)\in[0,+\infty)\times\mathcal{K},\\
&w^{div}|_{\partial\mathcal{K}}=0,\\
&(w^{div},\partial_{t}w^{div})(0,x)=(w_{0},w_{1})(x),\qquad\qquad\quad x\in\mathcal{K},\end{aligned}\right.

where
(w_{0},w_{1})
has a compact support and
\operatorname{supp}_{x}(G,G^{\theta},G^{\alpha})\subset\{x:|x|\leq t+M_{0}\}

(5.4)

\begin{split}&\langle{x}\rangle^{1/2}\langle{t-|x|}\rangle|\partial w^{div}|\lesssim\|(w_{0},w_{1})\|_{H^{9}(\mathcal{K})}+\sum_{|a|\leq 8}\sup_{y\in\mathcal{K}}|\partial^{a}F(0,y)|+\sup_{y\in\mathcal{K}}|G^{0}(0,y)|\\
&+\sum_{|a|\leq 8}\sup_{s\in[0,t]}\langle{s}\rangle\|\partial^{a}F(s)\|_{L^{2}(\mathcal{K}_{4})}+\sum_{i=1,2}\sup_{s\in[0,t]}\langle{s}\rangle^{3/2+\mu}\|G^{i}(s)\|_{L^{\infty}(\mathcal{K}_{3})}\\
&+\sup_{(s,y)\in[0,t]\times(\overline{\mathop{\mathbb{R}\kern 0.0pt}\nolimits^{2}\setminus\mathcal{K}_{2}})}\langle{y}\rangle^{1/2}\mathcal{W}_{3/2+\mu,1}(s,y)|G(s,y)|\\
