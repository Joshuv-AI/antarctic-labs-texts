Let

(3.7)

\displaystyle=U_{\tau}N^{(7)n}_{D}N^{(i)}_{D}Q_{T^{*}_{7}}h

\displaystyle=U_{\tau}N^{(7)n+1}_{D}N^{(7)*}_{D}N^{(i)}_{D}Q_{T^{*}_{7}}h

[

]

\displaystyle=N^{(7)^{\prime}n+1}_{D}V^{(7-i)*}_{T^{{}^{\prime}*}_{7}}Q_{T^{{}^{\prime}*}_{7}}\tau h\,\,(\text{by \eqref{tau, V^i*_T^*_7 commute}})

[
