\displaystyle m_{r}(q)=\min\biggl\{\tau_{2r}(q),\biggl(\frac{\tau(q)}{2}\biggr)^{2r-1},\frac{q}{2r}\biggr\}.

Lemma 2.1

(Burgess)
.

Let

be integers such that at least
r+1
of them are distinct. Let
f_{1}(x)=(x-b_{1})(x-b_{2})\cdots(x-b_{r})

. Let
\chi
be a primitive Dirichlet mod
q
. If

or

\left|\sum_{x\bmod{q}}\chi(f_{1}(x)\cdot f_{2}^{(\phi(q)-1)}(x))\right|\leq(4r)^{\omega(q)}\sqrt{q}(A_{j},q),

where
(A_{j},q)
is the greatest common divisor of

and

.
In particular,

\left|\sum_{x\bmod{q}}\chi(f_{1}(x)\cdot f_{2}^{(\phi(q)-1)}(x))\right|\leq\sum_{\begin{subarray}{c}j=1\\
A_{j}\neq 0\end{subarray}}^{2r}(4r)^{\omega(q)}\sqrt{q}(A_{j},q).

Definition 2.2
.

Define

\displaystyle s_{q}(r,B)=\sum_{1\leq b_{1}\leq B}\sum_{1\leq b_{2}\leq B}\cdots\sum_{1\leq b_{2r}\leq B}\min\bigl\{(A_{j},q)\colon 1\leq j\leq 2r,\,A_{j}\neq 0\bigr\}.

First, observe that from the trivial bound

\displaystyle s_{q}(r,B)\leq\sum_{1\leq b_{1}\leq B}\sum_{1\leq b_{2}\leq B}\cdots\sum_{1\leq b_{2r}\leq B}\sum_{\begin{subarray}{c}1\leq j\leq 2r\\
A_{j}\neq 0\end{subarray}}\left(A_{j},q\right)\leq B^{2r}q.

Now, the following lemmas give complementary upper bounds for this expression.

Lemma 2.3
.

Let
s_{q}(r,B)
be defined as in Definition
2.2
. Then

s_{q}(r,B)\leq 2r\bigl(\frac{\tau(q)}{2}\bigr)^{2r-1}B^{2r}.

Proof.

Using the bound

(A_{j},q)=\biggl(\prod_{\begin{subarray}{c}1\leq i\leq 2r\\
i\neq j\end{subarray}}(b_{i}-b_{j}),q\biggr)\leq\prod_{\begin{subarray}{c}1\leq i\leq 2r\\
i\neq j\end{subarray}}(b_{i}-b_{j},q),

and exchanging the order of summation, we have

(3)

\begin{split}s_{q}(r,B)\leq\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}1\leq b_{1},b_{2},\dots,b_{2r}\leq B\\
A_{j}\neq 0\end{subarray}}\left(A_{j},q\right)&\leq\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}1\leq b_{1},b_{2},\dots,b_{2r}\leq B\\
A_{j}\neq 0\end{subarray}}\prod_{\begin{subarray}{c}1\leq i\leq 2r\\
i\neq j\end{subarray}}(b_{i}-b_{j},q)\\
&=\sum_{1\leq j\leq 2r}\sum_{1\leq b_{j}\leq B}\sum_{\begin{subarray}{c}1\leq b_{1},\dots,b_{j-1},b_{j+1},\dots,b_{2r}\leq B\\
b_{j}\notin\{b_{1},\dots,b_{j-1},b_{j+1},\dots,b_{2r}\}\end{subarray}}\prod_{\begin{subarray}{c}1\leq i\leq 2r\\
i\neq j\end{subarray}}(b_{i}-b_{j},q)\\
&=\sum_{1\leq j\leq 2r}\sum_{1\leq b_{j}\leq B}\biggl(\sum_{\begin{subarray}{c}1\leq b\leq B\\
b\neq b_{j}\end{subarray}}(b-b_{j},q)\biggr)^{2r-1}\\
&=2r\sum_{1\leq a\leq B}\biggl(\sum_{\begin{subarray}{c}1\leq b\leq B\\
b\neq a\end{subarray}}(b-a,q)\biggr)^{2r-1}.\end{split}

Using the identity
\sum_{d\mid n}\phi(d)=n
, we write the inner sum as

\sum_{\begin{subarray}{c}1\leq b\leq B\\
b\neq a\end{subarray}}(b-a,q)=\sum_{\begin{subarray}{c}1\leq b\leq B\\
b\neq a\end{subarray}}\sum_{d\mid(b-a,q)}\phi(d)=\sum_{\begin{subarray}{c}d\mid q\\
d\leq B-1\end{subarray}}\phi(d)\sum_{\begin{subarray}{c}1\leq b\leq B\\
b\neq a\\
d\mid b-a\end{subarray}}1\leq\sum_{\begin{subarray}{c}d\mid q\\
d\leq B\end{subarray}}\phi(d)\frac{\lfloor B\rfloor}{d}=\lfloor B\rfloor\sum_{\begin{subarray}{c}d\mid q\\
d\leq B\end{subarray}}\frac{\phi(d)}{d},

where since the divisors of
q
pair up on either side of
\sqrt{q}
and
B<\sqrt{q}
,

(4)

\lfloor B\rfloor\sum_{\begin{subarray}{c}d\mid q\\
d\leq B\end{subarray}}\frac{\phi(d)}{d}\leq\lfloor B\rfloor\sum_{\begin{subarray}{c}d\mid q\\
d\leq B\end{subarray}}1\leq\frac{\lfloor B\rfloor}{2}\tau(q).

Putting inequalities (
3
) and (
4
) together,

\displaystyle s_{q}(r,B)

\displaystyle\leq 2r\sum_{1\leq a\leq B}\biggl(\sum_{\begin{subarray}{c}1\leq b\leq B\\
b\neq a\end{subarray}}(b-a,q)\biggr)^{2r-1}

\displaystyle\leq 2r\sum_{1\leq a\leq B}\biggl(\frac{\lfloor B\rfloor}{2}\tau(q)\biggr)^{2r-1}=2r\left(\frac{\tau(q)}{2}\right)^{2r-1}\lfloor B\rfloor^{2r},

as stated in the lemma.
∎

We bound the sum
s_{q}(r,B)
below without using the initial inequality on the
\gcd
in Lemma

Lemma 2.4
.

Let
s_{q}(r,B)
be defined as in Definition
2.2
. Then

s_{q}(r,B)\leq 2rB^{2r}\tau_{2r}(q).

Proof.

Exchanging the sum as before and using the identity
\sum_{d\mid n}\phi(d)=n
,

\displaystyle s_{q}(r,B)\leq\sum_{1\leq j\leq 2r}\mathop{\sum\nolimits}\limits_{\begin{subarray}{c}1\leq b_{1},b_{2},\dots,b_{2r}\leq B\\
A_{j}\neq 0\end{subarray}}(A_{j},q)

\displaystyle=\sum_{1\leq j\leq 2r}\mathop{\sum\nolimits}\limits_{\begin{subarray}{c}1\leq b_{1},b_{2},\dots,b_{2r}\leq B\\
A_{j}\neq 0\end{subarray}}\sum_{d\mid(A_{j},q)}\phi(d)

\displaystyle\leq\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}d\mid q\\
d\leq B^{2r-1}\end{subarray}}\phi(d)\sum_{\begin{subarray}{c}1\leq b_{1},b_{2},\dots,b_{2r}\leq B\\
A_{j}\neq 0\\
d\mid A_{j}\end{subarray}}1.

Let

be an ordered factorization of

with
d_{j}=1
, then

if

for each

. Every inner summand arises in this way for at least one such ordered factorization, therefore

\displaystyle\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}d\mid q\\
d\leq B^{2r-1}\end{subarray}}\phi(d)\sum_{\begin{subarray}{c}1\leq b_{1},b_{2},\dots,b_{2r}\leq B\\
A_{j}\neq 0\\
d\mid A_{j}\end{subarray}}1\leq\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}d\mid q\\

d_{j}=1\end{subarray}}\sum_{\begin{subarray}{c}1\leq b_{1}\leq B\\
b_{1}\neq b_{j}\\
d_{1}\mid(b_{1}-b_{j})\end{subarray}}\cdots\sum_{\begin{subarray}{c}1\leq b_{2r}\leq B\\
b_{2r}\neq b_{j}\\
d_{2r}\mid(b_{2r}-b_{j})\end{subarray}}1.

Note that there are

\displaystyle s_{q}(r,B)

\displaystyle\leq\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}d\mid q\\
d\leq B^{2r-1}\end{subarray}}\phi(d)\sum_{1\leq b_{j}\leq B}\sum_{\begin{subarray}{c}d_{1}\cdots d_{2r}=d\\
d_{j}=1\end{subarray}}\sum_{\begin{subarray}{c}1\leq b_{1}\leq B\\
b_{1}\neq b_{j}\\
d_{1}\mid(b_{1}-b_{j})\end{subarray}}\cdots\sum_{\begin{subarray}{c}1\leq b_{2r}\leq B\\
b_{2r}\neq b_{j}\\
d_{2r}\mid(b_{2r}-b_{j})\end{subarray}}1

(5)

\displaystyle\leq\sum_{1\leq j\leq 2r}\sum_{\begin{subarray}{c}d\mid q\\

Let

be positive integers, and let

Given an integer

|S_{\chi}(M,N)|\leq E_{q,r}(N),\quad\text{where }E_{q,r}(N)=CN^{1-1/r}q^{\frac{r+1}{4r^{2}}}(\log q)^{\frac{1}{2r}}T(q),

\bigl((4r)^{\omega(q)}m_{r}(q)\bigr)^{\left({\frac{1}{2r}}\right)}\bigl(\frac{q}{\phi(q)}\bigr)^{1/r}&\text{ under the conditions of Theorem }\ref{main theorem 2}.\end{cases}

We take

\displaystyle N\leq q^{\frac{r+1}{4r}}\bigl(C(\log q)^{\frac{1}{2r}}T(q)\bigr)^{r},
