ConjugacyClassesSubgroups2:= function(g)
Reset(GlobalMersenneTwister);

end;

Hcandidates:= function(G)
local Gcs,GN;
Gcs:=ConjugacyClassesSubgroups2(G);
GN:=NormalSubgroups(G);
return Filtered(List(Gcs,Representative),x->Length(Filtered(GN,y->IsSubgroup(x,y)))=1);;
end;

Norm1TorusJTransitiveGroup:= function(d,n)
local l,T,M;
l:=Concatenation(IdentityMat(d-1),[-List([1..d-1],One)]);
T:=TransitiveGroup(d,n);
M:=List(GeneratorsOfGroup(T),x->List([1..d-1],y->l[y^x]));
return Group(M);
end;

Norm1TorusJPermutationGroup:= function(G)
local d,l,T,M;
d:=NrMovedPoints(G);
l:=Concatenation(IdentityMat(d-1),[-List([1..d-1],One)]);
M:=List(GeneratorsOfGroup(G),x->List([1..d-1],y->l[y^x]));
return Group(M);
end;

Norm1TorusJCoset:= function(g,h)
local gg,hg,phg,d,l,M;
gg:=GeneratorsOfGroup(g);
hg:=SortedList(RightCosets(g,h));
phg:=List(gg,x->Permutation(x,hg,OnRight));
d:=Length(hg);
l:=Concatenation(IdentityMat(d-1),[-List([1..d-1],One)]);
M:=List(phg,x->List([1..d-1],y->l[y^x]));
return Group(M);
end;
