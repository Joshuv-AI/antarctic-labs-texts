Fuzzy Volterra Integral Equations with Piecewise Continuous Kernels: Theory and Numerical Solution
Samad Noeiaghdam, Aliona Dreglea, Denis N. Sidorov

Abstract
This study aims to discuss the existence and uniqueness of solution of fuzzy Volterra integral equations with piecewise continuous kernels. These types of problems are often encountered in balancing issues for systems with hereditary dynamics, such as electric load leveling. The method of successive approximations is applied and the main theorems are proved based on the method. Some examples are discussed and the results are presented for different values of μ \mu by plotting several graphs.

\mathcal{D}^{*}(F,y_{m})\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2(1-C_{t})}\omega_{\theta_{t-1},\theta_{t}}(Y,h)+\sum_{t=1}^{m^{\prime}}\frac{C_{t}^{m+1}L_{1}}{L(1-C_{t})}+\sum_{t=1}^{m^{\prime}}\frac{C_{t}^{2}+2C_{t}}{2LM_{t}(1-C_{t})}(L_{1}\omega_{s}(K_{t},h)+L_{2}\omega_{t}(K_{t},h))

where

\omega_{s}(K_{t},h)=\sup_{z_{1}\leq v\leq T\leq z_{2}}\{\sup|K_{t}(x,v)-K_{t}(y,v)|:|x-y|\leq h\},~t=1,2,...,m^{\prime},

and

\omega_{t}(K_{t},h)=\sup_{z_{1}\leq s\leq T\leq z_{2}}\{\sup|K_{t}(r,v_{1})-K_{t}(r,v_{2})|:|v_{1}-v_{2}|\leq h\},~t=1,2,...,m^{\prime}.

Proof:

We know

Z_{1}(v)=Y(v)\oplus(\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{0}(r))dr,~~~z_{1}\leq s,v\leq T\leq z_{2},

then

\begin{array}[]{l}\displaystyle\mathcal{D}(Z_{1}(v),y_{1}(v))=\mathcal{D}(Y(v),Y(v))\\
\\
\displaystyle+\mathcal{D}\bigg((\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{0}(r))dr,\\
\\
\displaystyle~~~~~~~~\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(v_{l},v)\odot G(Z_{0}(v_{t}))\oplus K_{t}(v_{l},v)\odot G(Z_{0}(v_{l}))\big]\bigg)\\
\\
\displaystyle=\mathcal{D}\bigg(\sum_{l=0}^{n-1}\sum_{t=1}^{m^{\prime}}(\mathcal{FR})\int_{\theta_{t-1}(v_{l})}^{\theta_{t}(v_{l})}K_{t}(r,v)\odot G(Y(r))dr,\\
\\
\displaystyle~~~~~~~~\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(v_{l},v)\odot G(Y(v_{l}))\oplus K_{t}(v_{l+1},v)\odot G(Y(v_{l+1}))\big]\bigg)\\
\\
\leq\displaystyle\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\mathcal{D}\bigg((\mathcal{FR})\int_{\theta_{t-1}(v_{l})}^{\theta_{t}(v_{l+1})}K_{t}(r,v)\odot G(Y(r))dr,\\
\\
\displaystyle~~~~~~~~\frac{h}{2}\odot\big[K_{t}(v_{l},v)\odot G(Y(v_{l}))\oplus K_{t}(v_{l+1},v)\odot G(Y(v_{l+1}))\big]\bigg)\\
\\
\leq\displaystyle\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\mathcal{D}\bigg((\mathcal{FR})\int_{\theta_{t-1}(v_{l})}^{\theta_{t}(v_{l+1})}K_{t}(r,v)\odot G(Y(r))dr,\\
\\
\displaystyle~~~~~~~~\frac{h}{2}\odot\big[K_{t}(r,v)\odot G(Y(v_{l}))\oplus K_{t}(r,v)\odot G(Y(v_{l+1}))\big]\bigg)\\
\\
\displaystyle~~~~~~~~+\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\mathcal{D}\bigg(\frac{h}{2}\odot\big[K_{t}(r,v)\odot G(Y(v_{l}))\oplus K_{t}(r,v)\odot G(Y(v_{l+1}))\big],\\
\\
\displaystyle~~~~~~~~\frac{h}{2}\odot\big[K_{t}(v_{l},v)\odot G(Y(v_{l}))\oplus K_{t}(v_{l+1},v)\odot G(Y(v_{l+1}))\big]\bigg)\end{array}

Applying the second part of the first corollary and regarding to Lemma 4 in [11] one have:

\begin{array}[]{ll}\displaystyle\mathcal{D}(Z_{1}(v),y_{1}(v))&\leq\displaystyle\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}|K_{t}(r,v)|\mathcal{D}\bigg((\mathcal{FR})\int_{\theta_{t-1}(v_{l})}^{\theta_{t}(v_{l+1})}G(Y(r))dr,\frac{h}{2}\odot\big[G(Y(v_{l}))\oplus G(Y(v_{l+1}))\big]\bigg)\\
\\
&\displaystyle~~~+\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\mathcal{D}\bigg(\frac{h}{2}\odot K_{t}(r,v)\odot G(Y(v_{l})),\frac{h}{2}\odot K_{t}(v_{l},v)\odot G(Y(v_{l}))\bigg)\\
\\
&\displaystyle~~~+\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\mathcal{D}\bigg(\frac{h}{2}\odot K_{t}(r,v)\odot G(Y(v_{l})),\frac{h}{2}\odot K_{t}(v_{l+1},v)\odot G(Y(v_{l+1}))\bigg)\\
\\
&\displaystyle\leq\frac{h}{2}\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}|K_{t}(r,v)|\omega_{[v_{l},v_{l+1}]}(G(Y),\frac{h}{2})\\
\\
&\displaystyle~~~+\frac{h}{2}\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}|K_{t}(r,v)-K_{t}(v_{r},v)|\mathcal{D}(G(Y(v_{l})),\tilde{0})\\
\\
&\displaystyle~~~+\frac{h}{2}\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}|K_{t}(r,v)-K_{t}(v_{l+1},v)|\mathcal{D}(G(Y(v_{l+1})),\tilde{0})\\
\\
&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{M_{t}(\theta_{t}-\theta_{t-1})}{2}\omega_{[\theta_{t-1},\theta_{t}]}(G(Y),h)+\sum_{t=1}^{m^{\prime}}(\theta_{t}-\theta_{t-1})M_{0}\omega_{s}(K_{t},h)\\
\\
&\displaystyle=\sum_{t=1}^{m^{\prime}}\frac{M_{t}(\theta_{t}-\theta_{t-1})}{2}\sup_{\theta_{t-1}\leq p,p_{1}\leq\theta_{t}}\{\mathcal{D}(G(Y(p)),G(Y(p_{1}))):|u-p_{1}|\leq h\}\\
\\
&\displaystyle~~~+\sum_{t=1}^{m^{\prime}}(\theta_{t}-\theta_{t-1})M_{0}\omega_{s}(K_{t},h)\\
\\
&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{M_{t}(\theta_{t}-\theta_{t-1})}{2}\sup_{\theta_{t-1}\leq p,p_{1}\leq\theta_{t}}\{L.\mathcal{D}(Y(p),Y(p_{1})):|u-p_{1}|\leq h\}\\
\\
&\displaystyle~~~+\sum_{t=1}^{m^{\prime}}(\theta_{t}-\theta_{t-1})M_{0}\omega_{s}(K_{t},h)\\
\\
&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{M_{t}L(\theta_{t}-\theta_{t-1})}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)+\sum_{t=1}^{m^{\prime}}(\theta_{t}-\theta_{t-1})M_{0}\omega_{s}(K_{t},h)\\
\end{array}

where
M_{0}=\sup_{\theta_{t-1}\leq s\leq\theta_{t}}\|G(Y(r))\|_{\digamma}
and
\omega_{s}(K_{t},h)
is the partial modules of continuity. Thus

\mathcal{D}(Z_{1}(v),y_{1}(v))\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}M_{0}\omega_{s}(K_{t},h).

We have

Z_{2}(v)=Y(v)\oplus(\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{1}(r))dr,

therefore

\begin{array}[]{l}\displaystyle\mathcal{D}(Z_{2}(v),y_{2}(v))=\displaystyle\mathcal{D}(Y(v),Y(v))\\
\\
+\mathcal{D}\bigg((\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{1}(r))dr,\\
\\
\displaystyle~~~~~~~\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(v_{l},v)\odot G(y_{1}(v_{l}))\oplus K_{t}(v_{l+1},v)\odot G(y_{1}(v_{l+1}))\big]\bigg)\\
\\
\displaystyle\leq\mathcal{D}\bigg((\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{1}(r))dr,\\
\\
\displaystyle~~~~~~~\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(r,v)\odot G(Z_{1}(v_{l}))\oplus K_{t}(r,v)\odot G(Z_{1}(v_{l+1}))\big]\bigg)\\
\\
\displaystyle+\mathcal{D}\bigg(\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(r,v)\odot G(Z_{1}(v_{l}))\oplus K_{t}(r,v)\odot G(Z_{1}(v_{l+1}))\big],\\
\\
\displaystyle~~~~~~~\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(r,v)\odot G(y_{1}(v_{l}))\oplus K_{t}(r,v)\odot G(y_{1}(v_{l+1}))\big]\bigg)\\
\\
\displaystyle+\mathcal{D}\bigg(\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(r,v)\odot G(y_{1}(v_{l}))\oplus K_{t}(r,v)\odot G(y_{1}(v_{l+1}))\big],\\
\\
\displaystyle~~~~~~~\sum_{t=1}^{m^{\prime}}\sum_{l=0}^{n-1}\frac{h}{2}\odot\big[K_{t}(v_{l},v)\odot G(y_{1}(v_{l}))\oplus K_{t}(v_{l+1},v)\odot G(y_{1}(v_{l+1}))\big]\bigg)\\
\\
\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{M_{t}(\theta_{t}-\theta_{t-1})}{2}\omega_{[\theta_{t-1},\theta_{t}]}(G(Z_{1}),\frac{h}{2})+\sum_{t=1}^{m^{\prime}}\frac{M_{t}(\theta_{t}-\theta_{t-1})}{2}\bigg[\mathcal{D}(G(Z_{1}(v_{l})),G(y_{1}(v_{l})))\\
\\
\displaystyle~~~~~~~+\mathcal{D}(G(Z_{1}(v_{l+1})),G(y_{1}(v_{l+1})))\bigg]+\sum_{t=1}^{m^{\prime}}(\theta_{t}-\theta_{t-1})M_{1}\omega_{s}(K_{t},h)\\
\\
\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Z_{1},h)+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\bigg[\mathcal{D}(Z_{1}(v_{l}),y_{1}(v_{l}))+\mathcal{D}(Z_{1}(v_{l+1}),y_{1}(v_{l+1}))\bigg]+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}M_{1}\omega_{s}(K_{t},h)\end{array}

where
M_{1}=\sup_{\theta_{t-1}\leq s\leq\theta_{t}}\|G(y_{1}(r))\|_{\digamma}
. Applying (
5
) and (
7
) and using induction for
m\geq 3
it can be concluded

\begin{array}[]{ll}\displaystyle\mathcal{D}(Z_{m}(v),y_{m}(v))&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Z_{m-1},h)+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\bigg[\mathcal{D}(Z_{m-1}(v_{l}),y_{m-1}(v_{l}))\\
\\
&\displaystyle+\mathcal{D}(Z_{m-1}(v_{l+1}),y_{m-1}(v_{l+1}))\bigg]+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}M_{m-1}\omega_{s}(K_{t},h)\end{array}

(8)

where
M_{m-1}=\sup_{\theta_{t-1}\leq v\leq\theta_{t}}\|G(y_{m-1}(v))\|_{\digamma}
. Taking supremum for
z_{1}\leq v\leq z_{2}
from (
8
) we have

\begin{array}[]{l}\displaystyle\mathcal{D}^{*}(Z_{m},y_{m})\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Z_{m-1},h)+\sum_{t=1}^{m^{\prime}}C_{t}\displaystyle\mathcal{D}^{*}(Z_{m-1},y_{m-1})+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}M_{m-1}\omega_{s}(K_{t},h),\\
\\
\displaystyle\mathcal{D}^{*}(Z_{m-1},y_{m-1})\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Z_{m-2},h)+\sum_{t=1}^{m^{\prime}}C_{t}\displaystyle\mathcal{D}^{*}(Z_{m-2},y_{m-2})+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}M_{m-2}\omega_{s}(K_{t},h),\\
\\
\vdots\\
\\
\displaystyle\mathcal{D}^{*}(Z_{1},y_{1})\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}M_{0}\omega_{s}(K_{t},h).\end{array}

(9)

If one multiple the above inequality to
1,C_{t},...,C_{t}^{m-1}
and find the summation then

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(Z_{m},y_{m})&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}\bigg(\omega_{[\theta_{t-1},\theta_{t}]}(Z_{m-1},h)+C_{t}\omega_{[\theta_{t-1},\theta_{t}]}(Z_{m-2},h)+...+C_{t}^{m-1}\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)\bigg)\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}\omega_{s}(K_{t},h)(M_{m-1}+C_{t}M_{m-2}+...+C_{t}^{m-1}M_{0}).\\
\end{array}

(10)

Moreover for
v_{1},v_{2}\in[z_{1},z_{2}]
with
|v_{1}-v_{2}\leq h|
one can write

\begin{array}[]{l}\mathcal{D}(Z_{m}(v_{1}),Z_{m}(v_{2}))\\
\\
\displaystyle=\mathcal{D}\bigg(Y(v_{1})\oplus\odot(\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v_{1})}^{\theta_{t}(v_{1})}K_{t}(r,v_{1})\odot G(Z_{m-1}(r))dr,\\
\\
\displaystyle~~~~~~~~~Y(v_{2})\oplus\odot(\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v_{2})}^{\theta_{t}(v_{2})}K_{t}(r,v_{2})\odot G(Z_{m-1}(r))dr\bigg)\\
\\
\displaystyle\leq\mathcal{D}(Y(v_{1}),Y(v_{2}))+\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v_{1})}^{\theta_{t}(v_{1})}|K_{t}(r,v_{1})-K_{t}(r,v_{2})|\mathcal{D}(G(Z_{m-1}(r)),\tilde{0})dr\\
\\
\displaystyle\leq\mathcal{D}(Y(v_{1}),Y(v_{2}))+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}\omega_{t}(K_{t},h)M^{\prime}_{m-1},\end{array}

where
\omega_{t}(K_{t},h)
is the partial modulus of continuity with respect to
t
. Let
M^{\prime}_{m-1}=\sup_{\theta_{t-1}\leq s\leq\theta_{t}}\|G(Z_{m-1}(r))\|_{\digamma}
then we can find the relation
Z_{m}
and
Y
as:

\omega_{[\theta_{t-1},\theta_{t}]}(Z_{m},h)\leq\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM_{t}}\omega_{t}(K_{t},h)M^{\prime}_{m-1}.

(11)

And if we substitute above inequality into (
10
) we obtain

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(Z_{m},y_{m})&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}(1+C_{t}+C_{t}^{2}+...+C_{t}^{m-1})\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2LM_{t}}\omega_{t}(K_{t},h)(C_{t}M^{\prime}_{m-2}+C_{t}^{2}M^{\prime}_{m-3}+...+C_{t}^{m-1}M^{\prime}_{0})\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM}\omega_{s}(K_{t},h)(M_{m-1}+C_{t}M_{m-2}+...+C_{t}^{m-1}M_{0}).\end{array}

(12)

Let
L_{1}=\max_{0\leq i\leq m-1}\{M_{i}\}
and
L_{2}=\max_{0\leq i\leq m-2}\{M^{\prime}_{i}\}
thus

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(Z_{m},y_{m})&\displaystyle\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2}(\frac{1-C_{t}^{m}}{1-C_{t}})\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{2LM_{t}}\omega_{t}(K_{t},h)(C_{t}+C_{t}^{2}+...+C_{t}^{m-1})L_{2}\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{LM}\omega_{s}(K_{t},h)(1+C_{t}+...+C_{t}^{m-1})L_{1}.\end{array}

(13)

From other hand
\frac{1-C_{t}^{m}}{1-C_{t}}\leq\frac{1}{1-C_{t}},t=1,2,...,m^{\prime}
for each
m\in N
thus

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(Z_{m},y_{m})&\displaystyle\leq\sum_{t=1}^{m^{\prime}}(\frac{C_{t}}{2(1-C_{t})})\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}^{2}+2C_{t}}{2LM_{t}(1-C_{t})}(L_{1}\omega_{s}(K_{t},h)+L_{2}\omega_{t}(K_{t},h)).\end{array}

Applying the inequality (
6
) we can write

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(F,y_{m})&\leq\mathcal{D}^{*}(F,Z_{m})+\mathcal{D}^{*}(Z_{m},y_{m})\\
\\
&\displaystyle\leq\sum_{t=1}^{m^{\prime}}(\frac{C_{t}^{m}}{1-C_{t}})\mathcal{D}^{*}(Z_{1},Z_{0})+\sum_{t=1}^{m^{\prime}}(\frac{C_{t}}{2(1-C_{t})})\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}^{2}+2C_{t}}{2LM_{t}(1-C_{t})}(L_{1}\omega_{s}(K_{t},h)+L_{2}\omega_{t}(K_{t},h)).\end{array}

Since

\begin{array}[]{ll}\displaystyle\mathcal{D}(Z_{1}(v),Z_{0}(v))&\displaystyle=\mathcal{D}\bigg(Y(v)\oplus(\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{0}(r))dr,Z_{0}(v)\bigg)\\
\\
&\displaystyle\leq\mathcal{D}\bigg((\mathcal{FR})\sum_{t=1}^{m^{\prime}}\int_{\theta_{t-1}(v)}^{\theta_{t}(v)}K_{t}(r,v)\odot G(Z_{0}(r))dr,\tilde{0}\bigg)\end{array}

we obtain

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(Z_{1},Z_{0})&\displaystyle\leq\sum_{t=1}^{m^{\prime}}M_{t}(\theta_{t}-\theta_{t-1})\sup_{\theta_{t-1}\leq s\leq\theta_{t}}\mathcal{D}(G(Y(r)),\tilde{0})\\
\\
&\displaystyle=\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{L}M_{0}\leq\sum_{t=1}^{m^{\prime}}\frac{C_{t}}{L}L_{1}.\end{array}

Thus we get

\begin{array}[]{ll}\displaystyle\mathcal{D}^{*}(F,y_{m})&\displaystyle\leq\sum_{t=1}^{m^{\prime}}(\frac{C_{t}}{2(1-C_{t})})\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)+\sum_{t=1}^{m^{\prime}}(\frac{C_{t}^{m+1}L_{1}}{L(1-C_{t})})\\
\\
&\displaystyle+\sum_{t=1}^{m^{\prime}}\frac{C_{t}^{2}+2C_{t}}{2LM_{t}(1-C_{t})}(L_{1}\omega_{s}(K_{t},h)+L_{2}\omega_{t}(K_{t},h)).\square\end{array}

Remark 1
.

As we know
C_{t}<1,t=1,2,...,m^{\prime}
and it shows
\lim_{m\rightarrow\infty}C_{t}^{m+1}=0,t=1,2,...,m^{\prime}
. And we have

\lim_{h\rightarrow 0}\omega_{[\theta_{t-1},\theta_{t}]}(Y,h)=0,~~\lim_{h\rightarrow 0}\omega_{s}(K_{t},h)=0,~~\lim_{h\rightarrow 0}\omega_{t}(K_{t},h)=0,~~t=1,2,...,m^{\prime}.

The convergence of this scheme can be obtained by
\lim_{m\rightarrow\infty,h\rightarrow 0}\mathcal{D}^{*}(F,y_{m})=0
.

6
Conclusion

In this work, the fuzzy Volterra integral equation of the second kind with piecewise kernel was studied. We applied the successive approximation scheme. The existence of an unique solution with the error bound and also the error estimation theorems were discussed. Some examples have been discussed. Plotting the graphs of fuzzy approximate solutions for various
\mu
and error functions we showed the accuracy of the method. As the future works, we will combine the method with the CESTAC-CADNA strategy to find the numerical optimality results and optimal distance.
