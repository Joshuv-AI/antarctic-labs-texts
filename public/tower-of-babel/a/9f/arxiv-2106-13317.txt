\displaystyle\big(x^{\alpha}y^{\prime}_{N,\varepsilon}(x)\big)^{\prime}=

\displaystyle\big(x^{\alpha}y^{\prime}_{N}(x)\big)^{\prime}[\text{\rm ln}_{N}(x)]^{-\varepsilon/2}+2x^{\alpha}y^{\prime}_{N}(x)\big([\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\big)^{\prime}

\displaystyle+x^{\alpha}y_{N}(x)\big([\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\big)^{\prime\prime}+\alpha x^{\alpha-1}y_{N}(x)\big([\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\big)^{\prime}.

(A.11)

Employing (
A.2
), (
A.4
), and (
A.5
), we next calculate several derivatives, which appear on the right-hand side of (
A.11
):

\displaystyle\big([\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\big)^{\prime}=(\varepsilon/2)x^{-1}[\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1},

(A.12)

\displaystyle\big([\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\big)^{\prime\prime}=-(\varepsilon/2)x^{-2}[\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}

\displaystyle\qquad\qquad\qquad\qquad+\big(\varepsilon^{2}\big/4\big)x^{-2}[\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-2}

(A.13)

\displaystyle\qquad\qquad\qquad\qquad+(\varepsilon/2)x^{-2}[\text{\rm ln}_{N}(x)]^{-\varepsilon/2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}\sum_{j=1}^{N}\prod_{\ell=1}^{j}[\text{\rm ln}_{\ell}(x)]^{-1},

\displaystyle\frac{y^{\prime}_{N}(x)}{y_{N}(x)}=(2x)^{-1}\Bigg[-1+\sum_{j=1}^{N}\prod_{\ell=1}^{j}[\text{\rm ln}_{\ell}(x)]^{-1}\Bigg].

(A.14)

It follows from the last equality in (
A.9
), from (
A.11
), from (
A.12
)–(
A.14
), and from (
3.34
) that

\displaystyle\frac{1}{y_{N,\varepsilon}(x)}\big(x^{\alpha}y^{\prime}_{N,\varepsilon}(x)\big)^{\prime}=q_{\alpha,N}(x)-(\varepsilon/2)x^{\alpha-2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}\Bigg(1-\sum_{j=1}^{N}\prod_{\ell=1}^{j}[\text{\rm ln}_{\ell}(x)]^{-1}\Bigg)

\displaystyle\qquad+(\varepsilon/2)x^{\alpha-2}\Bigg(-\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}+(\varepsilon/2)\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-2}

\displaystyle\qquad+\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}\sum_{j=1}^{N}\prod_{\ell=1}^{j}[\text{\rm ln}_{\ell}(x)]^{-1}\Bigg)+(\varepsilon/2)\alpha x^{\alpha-2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}.

\displaystyle\quad=q_{\alpha,N}(x)-(\varepsilon/2)(2-\alpha)x^{\alpha-2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}+\big(\varepsilon^{2}\big/4\big)x^{\alpha-2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-2}

\displaystyle\qquad+\varepsilon x^{\alpha-2}\prod_{k=1}^{N}[\text{\rm ln}_{k}(x)]^{-1}\sum_{j=1}^{N}\prod_{\ell=1}^{j}[\text{\rm ln}_{\ell}(x)]^{-1}

\displaystyle\quad=q_{\alpha,N,\varepsilon}(x).

(A.15)

Thus, the first equality in (
A.10
) is proved, and the second equality in (
A.10
)
is clear from the reduction of order approach.
