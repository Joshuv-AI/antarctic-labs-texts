Journal: DCDS-B
Zhe Jiao, Yong Xu

Abstract
This paper concerns about the stability for the acoustic wave motion with boundary frictions and random forces. We will show there exists a unique invariant measure for the stochastic evolution equation associated with this acoustic wave motion, and the invariant measure possesses the property of strong mixing. This result is new with respect to the literature on two accounts: (i) stochasticity is accounted for in the acoustic wave model; (ii) the controllability of the dynamical system modeling the acoustic wave motion implies the mixing.

&\leq C(T,\epsilon_{0})\{\int_{0}^{T}\int_{\Gamma_{1}}|\delta_{t}|^{2}d\Gamma_{1}dt+\int_{0}^{T}\int_{\Gamma_{1}}\kappa^{2}|\phi_{t}|^{2}d\Gamma_{1}dt\}+\textrm{LOT}(\Phi,\Psi).\end{aligned}

We estimate
\int_{\epsilon_{0}}^{T-\epsilon_{0}}\int_{\Gamma_{1}}\phi_{t}^{2}d\Gamma_{1}dt
and
\int_{0}^{T}\int_{\Gamma_{1}}\kappa^{2}|\phi_{t}|^{2}d\Gamma_{1}dt
. The boundary condition on
\Gamma_{1}
shows that

\displaystyle\begin{aligned} \int_{\epsilon_{0}}^{T-\epsilon_{0}}\int_{\Gamma_{1}}\phi_{t}^{2}d\Gamma_{1}dt\leq\int_{0}^{T}\int_{\Gamma_{1}}\kappa^{2}|\phi_{t}|^{2}d\Gamma_{1}dt\leq C(\rho,m,d,k)\int_{0}^{T}\int_{\Gamma_{1}}[\delta^{2}_{tt}+\delta^{2}_{t}+\delta^{2}]d\Gamma_{1}dt\end{aligned}

Then from (
14
) it deduces that

\displaystyle\begin{aligned} \int_{\epsilon_{0}}^{T-\epsilon_{0}}\mathcal{E}(t)dt\leq C(\tau_{1},h,T,\epsilon_{0},\rho,m,d,k)\int_{0}^{T}\int_{\Gamma_{1}}[\delta^{2}_{tt}+\delta^{2}_{t}+\delta^{2}]d\Gamma_{1}dt+C(h)\mathcal{E}(0)+\textrm{LOT}(\Phi,\Psi).\end{aligned}

(15)

From
\mathcal{E}(T)=\mathcal{E}(0)-\int_{0}^{T}\int_{\Gamma_{1}}\delta_{t}^{2}d\Gamma_{1}dt
, it follows that

\displaystyle\begin{aligned} &(T-2\epsilon_{0})[\mathcal{E}(0)-\int_{0}^{T}\int_{\Gamma_{1}}\delta_{t}^{2}d\Gamma_{1}dt]\\
&\leq(T-2\epsilon_{0})E(T)\\
&\leq\int_{\epsilon_{0}}^{T-\epsilon_{0}}\mathcal{E}(t)dt\\
&\leq C(\tau_{1},h,T,\epsilon_{0},\rho,m,d,k)\int_{0}^{T}\int_{\Gamma_{1}}[\delta^{2}_{tt}+\delta^{2}_{t}+\delta^{2}]d\Gamma_{1}dt+C(h)\mathcal{E}(0)+\textrm{LOT}(\Phi,\Psi),\end{aligned}

which implies

\displaystyle\begin{aligned} (T-2\epsilon_{0}-C(h))\mathcal{E}(0)\leq C(\tau_{1},h,T,\epsilon_{0},\rho,m,d,k)\int_{0}^{T}\int_{\Gamma_{1}}[\delta^{2}_{tt}+\delta^{2}_{t}+\delta^{2}]d\Gamma_{1}dt+\textrm{LOT}(\Phi,\Psi),\end{aligned}

(16)

Differentiating the boundary equation in time, multiplying
\delta_{t}
and integrating by parts, we have

\displaystyle\begin{aligned} 0&=\int_{0}^{T}\int_{\Gamma_{1}}\delta_{t}(\rho\phi_{tt}+m\delta_{ttt}+d\delta_{tt}+k\delta_{t})d\Gamma_{1}dt\\
&=[\rho\int_{\Gamma_{1}}\delta_{t}\phi_{t}d\Gamma_{1}]\bigg|_{0}^{T}-\rho\int_{0}^{T}\int_{\Gamma_{1}}\delta_{tt}\phi_{t}d\Gamma_{1}dt+[m\int_{\Gamma_{1}}\delta_{t}\delta_{tt}d\Gamma_{1}]\bigg|_{0}^{T}-m\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{tt}d\Gamma_{1}dt\\
&\qquad+[\frac{d}{2}\int_{\Gamma_{1}}\delta_{t}^{2}d\Gamma_{1}]\bigg|_{0}^{T}+k\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt,\end{aligned}

which implies

\displaystyle\begin{aligned} \int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{tt}d\Gamma_{1}dt&\leq C(\rho,m,d)\mathcal{E}(0)+\frac{k}{m}\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt+\frac{\rho}{2m}\int_{0}^{T}\int_{\Gamma_{1}}(\delta_{tt}+\phi_{t})^{2}d\Gamma_{1}dt\\
&\leq C(\rho,m,d)\mathcal{E}(0)+\frac{k}{m}\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt+\frac{\rho}{2m}\int_{0}^{T}\int_{\Gamma_{1}}(\delta_{t}+\delta)^{2}d\Gamma_{1}dt\\
&\leq C(\rho,m,d)\mathcal{E}(0)+\frac{k+\rho}{m}\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt+\frac{\rho}{m}\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}d\Gamma_{1}dt.\end{aligned}

(17)

Multiplying
\delta
and integrating by parts, we have

\displaystyle\begin{aligned} 0&=\int_{0}^{T}\int_{\Gamma_{1}}\delta(\rho\phi_{t}+m\delta_{tt}+d\delta_{t}+k\delta)d\Gamma_{1}dt\\
&=[\rho\int_{\Gamma_{1}}\delta\phi d\Gamma_{1}]\bigg|_{0}^{T}-\rho\int_{0}^{T}\int_{\Gamma_{1}}\delta_{t}\phi d\Gamma_{1}dt+[m\int_{\Gamma_{1}}\delta\delta_{t}d\Gamma_{1}]\bigg|_{0}^{T}-m\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt\\
&\qquad+[\frac{d}{2}\int_{\Gamma_{1}}\delta^{2}d\Gamma_{1}]\bigg|_{0}^{T}+k\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}d\Gamma_{1}dt,\end{aligned}

which implies

\displaystyle\begin{aligned} \int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}d\Gamma_{1}dt&\leq C(\rho,m,d,k)\mathcal{E}(0)+\tau_{2}\int_{0}^{T}\int_{\Gamma_{1}}\phi^{2}d\Gamma_{1}dt+(C(\tau_{2},\rho)+m)\int_{0}^{T}\int_{\Gamma_{1}}\delta_{t}^{2}d\Gamma_{1}dt\\
&\leq[C(\rho,m,d,k)+\tau_{2}T]\mathcal{E}(0)+(C(\tau_{2},\rho)+\frac{m}{k})\int_{0}^{T}\int_{\Gamma_{1}}\delta_{t}^{2}d\Gamma_{1}dt.\end{aligned}

(18)

Then from (
16
) and choosing
T_{0}:=\frac{2\epsilon_{0}+C(h)+C(\rho,m,d,k)}{1-\tau_{2}}
, combining (
17
) and (
18
) gives

\displaystyle\begin{aligned} (T-T_{0})\mathcal{E}(0)\leq C(\tau_{1},\tau_{2},h,T,\epsilon_{0},\rho,m,d,k)\int_{0}^{T}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt+\textrm{LOT}(\Phi,\Psi).\end{aligned}

(19)

Claim that there exists a constant
C_{T}>0
such that the solution of (
8
) satisfies the inequality

\displaystyle\textrm{LOT}(\Phi,\Psi)\leq C_{T}\|\delta_{t}\|^{2}_{L^{2}([0,T]\times\Gamma_{1})}.

(20)

We use the method of contradiction. Suppose that this claim is false. Then there exists a sequence of solutions
\{\Phi^{(n)}(t),\Psi^{(n)}(t)\}
to (
8
) with the initial data

\{\Phi^{(n)}(0),\Psi^{(n)}(0)\}\subset\mathcal{H}
such that

\displaystyle\begin{aligned} \textrm{LOT}(\Phi^{(n)},\Psi^{(n)})=1,\quad\|\delta^{(n)}_{t}\|^{2}_{L^{2}([0,T]\times\Gamma_{1})}\xrightarrow{n\rightarrow\infty}0.\end{aligned}

(21)

From (
19
), it follows that
\|\{\Phi^{(n)}(0),\Psi^{(n)}(0)\}\|_{\mathcal{H}}
is bounded. Then there is a subsequence, still denoted by
\{\Phi^{(n)}(0),\Psi^{(n)}(0)\}
, and
(\hat{\Phi}(0),\hat{\Psi}(0))\in\mathcal{H}
such that

\displaystyle\{\Phi^{(n)}(0),\Psi^{(n)}(0)\}\xrightarrow{weakly}\{\hat{\Phi}(0),\hat{\Psi}(0)\},\quad\textrm{in $\mathcal{H}$}

as
n\rightarrow\infty
. Let
\{\hat{\Phi}(t),\hat{\Psi}(t)\}
be the solution of (
8
) subject to the initial data
\{\hat{\Phi}(0),\hat{\Psi}(0)\}
. By (
4
), we see that
\|\{\Phi^{(n)}(t),\Psi^{(n)}(t)\}\|_{C([0,T];\mathcal{H})}
is bounded, and then

\displaystyle\{\Phi^{(n)}(t),\Psi^{(n)}(t)\}\xrightarrow{\textrm{weak star}}\{\hat{\Phi}(t),\hat{\Psi}(t)\},\quad\textrm{in $L^{\infty}([0,T];\mathcal{H})$}.

(22)

Let
\mathcal{N}=H^{-\eta}(D)\times(H^{1}(D))^{\prime}\times H^{-\eta}(\Gamma_{1})\times H^{-\eta}(\Gamma_{1})
. For any
\omega\in H^{1}
and
t\in(0,T)
, we have the following equation holds true

(\phi^{(n)}_{tt},\omega)_{H^{-1}\times H^{1}}=\int_{D}\Delta\phi^{(n)}\omega dx=\int_{\Gamma_{1}}\delta^{(n)}_{t}\omega d\Gamma_{1}-\int_{D}\nabla\phi^{(n)}\cdot\nabla\omega dx,

which implies that
|(\phi^{(n)}_{tt},\omega)_{H^{-1}\times H^{1}}|\leq C(\|\delta^{(n)}\|_{L^{2}},\|\nabla\phi^{(n)}\|_{L^{2}})\|\omega\|_{H^{1}}
. Hence
\phi^{(n)}_{tt}\in L^{\infty}([0,T],(H^{1}(D))^{\prime})
. Thus, it follows that
\|\{\Phi^{(n)}(t),\Psi^{(n)}(t)\}\|_{L^{\infty}([0,T],\mathcal{N})}
is bounded uniformly. Due to
\mathcal{H}\subset\subset\mathcal{M}\subset\mathcal{N}
, we deduce from the Aubin’s theorem
[
22
]
that

\displaystyle\{\Phi^{(n)}(t),\Psi^{(n)}(t)\}\xrightarrow{\textrm{strongly}}\{\hat{\Phi}(t),\hat{\Psi}(t)\},\quad\textrm{in $L^{\infty}([0,T];\mathcal{M})$},

which gives

\displaystyle\|\{\hat{\Phi}(t),\hat{\Psi}(t)\}\|_{C([0,T];\mathcal{M})}=\textrm{LOT}(\Phi^{(n)},\Psi^{(n)})=1.

(23)

By (
21
) and (
22
), we have
\hat{\delta}_{t}=0
. Let
P=\hat{\phi}_{tt}
. Then we have

\left\{\begin{array}[]{ll}P_{tt}=c^{2}\triangle P,\quad\textrm{in $\mathbb{R}^{+}\times D$.}\\
\frac{\partial P}{\partial\mathbf{n}}=0,\quad\textrm{on $\mathbb{R}^{+}\times\Gamma_{1}$}\\
P=0,\quad\textrm{on $\mathbb{R}^{+}\times\Gamma_{0}$}.\end{array}\right.

By Holmgren’s Uniqueness Theorem, choosing
T>2\textrm{diam}(D)
, it follows that
P=\hat{\phi}_{tt}=0
. And then we obtain

\left\{\begin{array}[]{ll}\triangle\hat{\phi}=0,\quad\textrm{in $D$.}\\
\frac{\partial\hat{\phi}}{\partial\mathbf{n}}=0,\quad\textrm{on $\Gamma_{1}$}\\
\hat{\phi}=0,\quad\textrm{on $\Gamma_{0}$},\end{array}\right.

which implies that
\hat{\phi}=0
. The boundary condition shows that
\hat{\delta}=0
. Then we obtain
\{\hat{\Phi}(t),\hat{\Psi}(t)\}=0
which contradicts (
23
). Therefore, we prove the inequality (
20
). ∎

3
Main Result

The following theorem is the main result of this paper.

Theorem 1
(Mixing)

Assume that
\mathbf{x}\in\mathcal{H}
is
\mathcal{F}_{0}
-measurable and
\mathbb{E}\|\mathbf{x}\|_{\mathcal{H}}^{2}\leq\infty
. Let
X(t,\mathbf{x})
be the mild solution of system (
7
) with the initial data
\mathbf{x}
. There exists a unique invariant measure
\mu
for the system (
7
) which is strong mixing, that is,

\lim_{t\rightarrow\infty}\mathbb{E}f(X(t,\mathbf{x}))=\int_{\mathcal{H}}f(z)\mu(dz),\quad\textrm{strongly in $L^{2}(\mathcal{H},\mu)$}

for any
f\in C_{b}(\mathcal{H})
.

It shows that the limit of the average of
f(X(t,\mathbf{x}))
is a quantity that does not depend on the initial point. Next, we will give the proof of this main result.

Proof 3

The subsequent proof consists of several steps.

Step 1.

Let
P_{t}
stand for the transition function of the family defined as the law of
X(t,\mathbf{x})
under the probability measure
\mathbb{P}

P_{t}(\mathbf{x},\Lambda)=\mathbb{P}(X(t,\mathbf{x})\in\Lambda),\quad\Lambda\in\mathcal{B}(\mathcal{H}),\quad t>0.

The corresponding Markov semigroups are given by

\displaystyle\begin{aligned} &M_{t}:C(\mathcal{H})\rightarrow C(\mathcal{H}),\quad M_{t}f(\mathbf{x})=\int_{\mathcal{H}}P_{t}(\mathbf{x},dz)f(z),\\
&M^{\ast}_{t}:\mathcal{P}(\mathcal{H})\rightarrow\mathcal{P}(\mathcal{H}),\quad M^{\ast}_{t}\lambda(\Lambda)=\int_{\mathcal{H}}P_{t}(\mathbf{x},\Lambda)\lambda(d\mathbf{x}).\end{aligned}

And these two semigroups satisfy the duality relation

(M_{t}f,\lambda)=(f,M_{t}^{\ast}\lambda)

for any
f\in C(\mathcal{H})
,
\lambda\in\mathcal{P}(\mathcal{H})
.

Due to
\frac{1}{2}\|X(t,\mathbf{x})\|^{2}_{\mathcal{H}}=\mathcal{E}(t)
, then from Young’s inequality it follows that

\displaystyle\begin{aligned} \|X(t,\mathbf{x})\|^{2}_{\mathcal{H}}&=\|\mathbf{x}\|^{2}_{\mathcal{H}}-2d\int_{0}^{t}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt-2\int_{\Gamma_{1}}\int_{0}^{t}\delta_{t}dW(s)d\Gamma_{1}\\
&\leq\|\mathbf{x}\|^{2}_{\mathcal{H}}-2d\int_{0}^{t}\int_{\Gamma_{1}}\delta^{2}_{t}d\Gamma_{1}dt+2d\int_{\Gamma_{1}}\big(\int_{0}^{t}\delta_{t}dW(s)\big)^{2}d\Gamma_{1}+\frac{|\Gamma_{1}|}{4d}.\end{aligned}

(24)

By the Itô isometry
\int_{\Gamma_{1}}\mathbb{E}\big(\int_{0}^{t}\delta_{t}dW(s)\big)^{2}d\Gamma_{1}=\int_{\Gamma_{1}}\mathbb{E}\big(\int_{0}^{t}\delta_{t}ds\big)^{2}d\Gamma_{1}
, we have from (
24
)

\displaystyle\mathbb{E}\|X(t,\mathbf{x})\|^{2}_{\mathcal{H}}\leq\mathbb{E}\|\mathbf{x}\|^{2}_{\mathcal{H}}+\frac{|\Gamma_{1}|}{4d}.

(25)

Define the linear operator

Q_{t}\mathbf{x}=\int_{0}^{t}S(s)BB^{\ast}S^{\ast}(s)\mathbf{x}ds,\quad\mathbf{x}\in\mathcal{H}.

Due to
\sup\limits_{t>0}\mathbb{E}\|X(t,0)\|^{2}_{\mathcal{H}}=\sup\limits_{t\geq 0}\textrm{Tr}Q_{t}
, from (
25
) we get
\sup\limits_{t\geq 0}\textrm{Tr}Q_{t}\leq\infty
. Hence by Theorem 6.2.1 in
[
19
]
there exists an invariant measure
\mu
for system (
7
), that is,
M_{t}^{\ast}\mu=\mu
.

Step 2.

Note that
X(t,\mathbf{x})
is a Gaussian random variable with mean
S(t)\mathbf{x}
and covariance
Q_{t}
.
From the Doob’s theorem
[
23
]
, if the transition function
P_{t}(\mathbf{x},\cdot)
defined on
\mathcal{B}(\mathcal{H})
is
t_{0}
-regular for some
t_{0}>0
, then
\mu
is the unique invariant measure for
P_{t}
, which is also strong mixing. Since
X(t)
is Gaussian, the regularity of
P_{t}
is a consequence of the strong Feller property.

Assume that the range of
S(t)
is a subset of the range of
Q_{t}^{\frac{1}{2}}
, that is,

\textrm{Ran}S(t)\subset\textrm{Ran}Q_{t}^{\frac{1}{2}}
,
t>0
.
Then we can denote a linear bounded operator
R(t)=Q_{t}^{-\frac{1}{2}}\circ S(t)
. By the Cameron-Martin formula, we have for any
\tilde{\mathbf{x}}\in\mathcal{H}

\displaystyle\frac{dP_{t}(\mathbf{x},\cdot)}{dP_{t}(0,\cdot)}(\tilde{\mathbf{x}})=F(t,\mathbf{x},\tilde{\mathbf{x}})

(26)

with

\displaystyle\begin{aligned} F(t,\mathbf{x},\tilde{\mathbf{x}})&=\exp\{\big(Q_{t}^{-\frac{1}{2}}\circ S(t)\mathbf{x},Q_{t}^{-\frac{1}{2}}\tilde{\mathbf{x}}\big)_{\mathcal{H}}-\frac{1}{2}\|Q_{t}^{-\frac{1}{2}}\circ S(t)\mathbf{x}\|_{\mathcal{H}}^{2}\}.\end{aligned}

Let
\{e_{k}(t)\}
be a complete orthonormal basis on
\mathcal{H}
which diagonalizes
Q_{t}
, and
\{\eta_{k}(t)\}
the corresponding set of eigenvalues of
Q_{t}
. Set

G_{N}(t,\mathbf{x},\tilde{\mathbf{x}})=\exp\{\sum_{k=1}^{N}\big[\frac{\big(S(t)\mathbf{x},e_{k}(t)\big)\big(\tilde{\mathbf{x}},e_{k}(t)\big)}{\eta_{k}(t)}-\frac{1}{2}\frac{|\big(S(t)\mathbf{x},e_{k}(t)\big)|^{2}}{\eta_{k}(t)}\big]\}.

It is easy to check that for any
\tilde{\mathbf{x}}\in\mathcal{H}
we have
\lim\limits_{N\rightarrow\infty}G_{N}(t,\mathbf{x},\tilde{\mathbf{x}})=F(t,\mathbf{x},\tilde{\mathbf{x}})
almost surly in
L^{2}(\mathcal{H},\mathcal{B}(\mathcal{H}),P_{t}(0,\cdot))
. If
\mathbf{h}\in\mathcal{H}
, we have

\displaystyle\big(\frac{d}{d\mathbf{x}}G_{N}(t,\mathbf{x},\tilde{\mathbf{x}}),\mathbf{h}\big)=G_{N}(t,\mathbf{x},\tilde{\mathbf{x}})\sum_{k=1}^{N}\big[\frac{\big(S(t)\mathbf{h},e_{k}(t)\big)\big(\tilde{\mathbf{x}},e_{k}(t)\big)}{\eta_{k}(t)}-\frac{\big(S(t)\mathbf{x},e_{k}(t)\big)\big(S(t)\mathbf{h},e_{k}(t)\big)}{\eta_{k}(t)}\big],

which implies

\displaystyle\begin{aligned} \lim_{N\rightarrow\infty}\big(\frac{d}{d\mathbf{x}}G_{N}(t,\mathbf{x},\tilde{\mathbf{x}}),\mathbf{h}\big)&=F(t,\mathbf{x},\tilde{\mathbf{x}})\big\{\big(Q_{t}^{-\frac{1}{2}}\circ S(t)\mathbf{h}_{1},Q_{t}^{-\frac{1}{2}}\tilde{\mathbf{x}}\big)-\big(Q_{t}^{-\frac{1}{2}}\circ S(t)\mathbf{x},Q_{t}^{-\frac{1}{2}}\circ S(t)\mathbf{h}\big)\big\}\\
&=F(t,\mathbf{x},\tilde{\mathbf{x}})(R(t)\mathbf{h},Q_{t}^{-\frac{1}{2}}\tilde{\mathbf{x}}-R(t)\mathbf{x}\big).\end{aligned}

(27)

For any
f(\mathbf{x})\in B_{b}(\mathcal{H})
, combining (
27
) and (
26
) gives

\displaystyle\begin{aligned} \big(\frac{d}{d\mathbf{x}}M_{t}f(\mathbf{x}),\mathbf{h}\big)&=\big(\frac{d}{d\mathbf{x}}\int_{\mathcal{H}}P_{t}(\mathbf{x},dz)f(z),\mathbf{h}\big)\\
&=\frac{d}{d\mathbf{x}}\big(\int_{\mathcal{H}}F(t,\mathbf{x},z)P_{t}(0,dz)f(z),\mathbf{h}\big)\\
&=\int_{\mathcal{H}}\big(R(t)\mathbf{h},Q_{t}^{-\frac{1}{2}}z-R(t)\mathbf{x}\big)F(t,\mathbf{x},\tilde{\mathbf{x}})P_{t}(0,dz)f(z)\\
&=\int_{\mathcal{H}}\big(R(t)\mathbf{h},Q_{t}^{-\frac{1}{2}}z-R(t)\mathbf{x}\big)P_{t}(\mathbf{x},dz)f(z)\\
&=\int_{\mathcal{H}}\big(R(t)\mathbf{h},Q_{t}^{-\frac{1}{2}}z\big)P_{t}(0,dz)f(S(t)\mathbf{x}+z).\end{aligned}

(28)

Then it deduce from (
28
) that for any
\mathbf{x}_{1}
,
\mathbf{x}_{2}\in\mathcal{H}
and
\tau\in[0,1]

\displaystyle\begin{aligned} \|M_{t}f(\mathbf{x}_{1})-M_{t}f(\mathbf{x}_{2})\|&=\big(\frac{d}{d\mathbf{x}}M_{t}f(\mathbf{x}_{1}+\tau(\mathbf{x}_{2}-\mathbf{x}_{1}),\mathbf{x}_{2}-\mathbf{x}_{1}\big)\\
&=\int_{\mathcal{H}}\big(R(t)(\mathbf{x}_{2}-\mathbf{x}_{1}),Q_{t}^{-\frac{1}{2}}z\big)P_{t}(0,dz)f(S(t)(\mathbf{x}_{1}+\tau(\mathbf{x}_{2}-\mathbf{x}_{1}))+z)\\
&\leq\sup_{\mathbf{x}\in\mathcal{H}}|f(\mathbf{x})|\big[\int_{\mathcal{H}}|\big(R(t)(\mathbf{x}_{2}-\mathbf{x}_{1}),Q_{t}^{-\frac{1}{2}}z\big)|^{2}P_{t}(0,dz)\big]^{\frac{1}{2}}\\
&\leq\sup_{\mathbf{x}\in\mathcal{H}}|f(\mathbf{x})|\|R(t)\|\|\mathbf{x}_{2}-\mathbf{x}_{1}\|_{\mathcal{H}}.\end{aligned}

(29)

Since
f(\mathbf{x})\in B_{b}(\mathcal{H})
and
R(t)
is a bounded operator, we conclude from (
29
) that
P_{t}(\mathbf{x},\cdot)
is a strong Feller semigroup.

Step 3.

In this step, we will prove
\textrm{Ran}S(t)\subset\textrm{Ran}Q_{t}^{\frac{1}{2}}
. Considering the deterministic system

\left\{\begin{array}[]{ll}d\mathbf{Y}(t)=A(\mathbf{Y}(t))dt+B\mathbf{v},\\
\mathbf{Y}(0)=\mathbf{y}_{0}.\end{array}\right.

(30)

Let
T>0
. The solution of (
30
) can be written as

\displaystyle\mathbf{Y}(t)=S(t)\mathbf{y}_{0}+\int_{0}^{t}S(t-s)B\mathbf{v}ds,\quad t\in[0,T].

Define the following operator

F_{T}:L^{2}([0,T];\mathcal{L})\rightarrow\mathcal{H},\quad F_{T}\mathbf{v}=\int_{0}^{T}S(T-s)B\mathbf{v}ds.

Note that Ran
F_{T}
consists of all states reachable in time
T
from zero. Let
Z_{T}\in\textrm{Dom}(A^{\ast})
. Let us also recall that
\textrm{Dom}(A^{\ast})
is dense in
\mathcal{H}
. Then for
\mathbf{v}\in L^{2}([0,T];\mathcal{L})
we have

\displaystyle\begin{aligned} (\mathbf{v},F^{\ast}_{T}(Z_{T}))_{L^{2}([0,T];\mathcal{L})}=(F_{T}(\mathbf{v}),Z_{T})_{\mathcal{H}}=\int_{0}^{T}(\mathbf{v},B^{\ast}S^{\ast}(T-s)Z_{T})_{\mathcal{H}}ds,\end{aligned}

which implies that

(F^{\ast}_{T}\mathbf{x})(s)=B^{\ast}S^{\ast}(T-s)\mathbf{x},\quad\mathbf{x}\in\mathcal{H},s\in[0,T].

By the definition of
Q_{T}^{\frac{1}{2}}
, we see that
Q_{T}=F_{F}F^{\ast}_{T}
, and
\textrm{Ran}Q_{T}^{\frac{1}{2}}=\textrm{Ran}F_{T}
.

Due to
\mathbf{Y}(T)=S(t)\mathbf{y}_{0}+F_{T}\mathbf{v}
, system (
30
) is null controllable in time
T
, which means that given any
\mathbf{y}_{0}\in\mathcal{H}
, there exists a control function
\mathbf{v}\in L^{(}[0.T];\mathcal{L})
such that
\mathbf{Y}(T)=0
, if and only if

\textrm{Ran}S(t)\subset\textrm{Ran}F_{T}
.
Therefore, to prove
\textrm{Ran}S(t)\subset\textrm{Ran}Q_{t}^{\frac{1}{2}}
, it remains to show that system (
30
) is null controllable in time
T
.

From
[
24
, Theorem 2.44]
, the system (
30
) is null controllable in time
T
if and only if there exists a positive constant
C_{\textrm{obs}}>0
such that

\displaystyle\int_{0}^{T}\|B^{\ast}S^{\ast}(t)Z\|^{2}_{\mathcal{L}}ds\geq C_{\textrm{obs}}\|S^{\ast}(T)Z\|^{2}_{\mathcal{H}},\quad\forall Z\in\textrm{Dom}(A^{\ast}).

(31)

Let
Z(t)
be the solution of the adjoint system

\left\{\begin{array}[]{ll}dZ(t)=A^{\ast}(Z(t))dt,\quad t\in[0,T]\\
Z(T)=Z_{T}\in\textrm{Dom}(A^{\ast}).\end{array}\right.

Then the inequality (
31
) is equivalent to

\displaystyle\int_{0}^{T}\|B^{\ast}Z(t)\|^{2}_{\mathcal{L}}ds\geq C_{\textrm{obs}}\|Z(0)\|^{2}_{\mathcal{H}},

which is a consequence of Lemma
2
.

This concludes the proof of our main theorem.∎
