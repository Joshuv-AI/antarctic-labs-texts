Abstract
In this paper, we will use the normalized intetral Ricci curvature to investigate Liouville type property of p p harmonic function on Riemannian manifold. secondly, we will use the BiRic curvature to obtian Liuville theorem for p p harmonic function or p p harmonic 1 form. Lastly, we we will use the BiRic curvature to obtian Liouville theorem forharmonic q ⁡ ( q ≥ 2 ) q(q\geq 2) form.

\|\nabla\omega\|^{2}=\sum\nabla_{r}\omega_{i_{1}\cdots i_{q}}\nabla^{r}\omega^{i_{1}\cdots i_{q}}

\begin{split}\frac{1}{2}\Delta\|\omega\|^{2}&=\|\nabla\omega\|^{2}+\sum\omega_{i_{1}\cdots l_{q}}\nabla^{r}\nabla_{r}\omega^{i_{1}\cdots i_{q}}\\
&=\|\nabla\omega\|^{2}+\sum R_{i_{s}}^{r}\omega_{i_{1}\cdots r\cdots i_{q}}\omega^{i_{1}\cdots i_{q}}-\sum_{t<s}^{1\cdots q}R_{{i_{t}}i_{s}}^{vu}\omega_{i_{1}\cdots v\cdots u\cdots i_{p}}\omega^{i_{1}\cdots i_{q}}\\
&=\|\nabla\omega\|^{2}+q\sum R_{i_{1}}^{r}\omega_{ri_{2}\cdots i_{q}}\omega^{i_{1}\cdots i_{q}}-\sum_{t<s}^{1\cdots p}R_{i_{t}i_{s}}^{vu}\omega_{i_{1}\cdots v\cdots u\cdots i_{q}}\omega^{i_{1}\cdots i_{q}}\\
&=\|\nabla\omega\|^{2}+q\sum R_{ij}\omega_{i_{2}\cdots i_{q}}^{i}\omega_{ji_{2}\cdots i_{q}}-\frac{q(q-1)}{2}\sum R_{tjih}\omega_{i_{3}\cdots i_{q}}^{tj}\omega^{ih{i_{3}\cdots i_{q}}}+\Delta\omega.\end{split}

(5.1)

On the other hand, we have

\frac{1}{2}\Delta\|\omega\|^{2}=\|\omega\|\Delta\|\omega\|+\|\nabla\|\omega\|\|^{2}\\
=\|\omega\|\Delta\|\omega\|+\|\nabla\omega\|^{2}-F(\omega),

We take cut off function

(5.2)

\begin{split}&\int_{M}|\nabla h|^{2}|\omega|^{2}\mathrm{d}v_{g}\\

&\geq\int_{M}|h|^{2}\bigg(F(\omega)+\Phi_{2}+2(|A|^{2}+\overline{Ric}(n))|\omega|^{2}\bigg)\mathrm{d}v_{g}.\end{split}

(5.3)

where

\begin{split}\Phi_{2}=&2\sum R_{ij}\omega_{i_{2}}^{i}\omega_{ji_{2}}-\sum R_{kjih}\omega^{k{}{j}}\omega^{ih{}}\\
=&2\sum R_{ij}\omega_{i_{2}}^{i}\omega_{ji_{2}}-\sum R_{kjih}\omega^{k{}{j}}\omega^{ih{}}.\end{split}

Notice that

\begin{split}&\Phi_{2}+2(|A|^{2}+\overline{Ric}(n))|\omega|^{2}\\

&+2(|A|^{2}+\overline{Ric}(n))|\omega|^{2p-2}\\

&-\sum\big(\overline{R}_{kjih}+A_{ki}A_{jh}-A_{kh}A_{ji}\big)\omega^{k{}{j}}\omega^{ih{}}+q(|A|^{2}+\overline{Ric}(n))|\omega|^{2}.\end{split}

(5.5)

Since

Theorem 5.2
.

\begin{split}&q\sum R_{ij}\omega_{i_{2}\cdots i_{q}}^{i}\omega_{ji_{2}\cdots i_{q}}-\frac{q(q-1)}{2}\sum R_{kjih}\omega_{i_{3}\cdots i_{q}}^{k{}{j}}\omega^{ih{i_{3}\cdots i_{q}}}\\
&+(|A|^{2}+\overline{Ric}(n))|\omega|^{2p-2}\\

&-\frac{q(q-1)}{2}\sum\big(\overline{R}_{kjih}+A_{ki}A_{jh}-A_{kh}A_{ji}\big)\omega_{i_{3}\cdots i_{q}}^{k{}{j}}\omega^{ih{i_{3}\cdots i_{q}}}+q(|A|^{2}+\overline{Ric}(n))|\omega|^{2p-2}.\end{split}

Since

, we have

\begin{split}Ric(X,X)+Ric(n)-K(X,n)\geq 0.\end{split}
