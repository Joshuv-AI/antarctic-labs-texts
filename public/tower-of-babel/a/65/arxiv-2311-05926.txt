\displaystyle=\mathbb{P}\Bigg(\int_{0}^{\infty}e^{\rho_{1}W(s)-(\lambda_{1}+\Lambda)(\mu-1)s}ds<\frac{J^{1-\mu}(0)}{(\mu-1)\mathopen{}\mathclose{{\left(\frac{1}{2}\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{\mu}{\mu-p}}(x)dx}}\right)^{\frac{p-\mu}{p}}+\delta\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{n}{n-1}}(x)dx}}\right)^{1-n}}}\right)}\Bigg)

\displaystyle=\mathbb{P}\Bigg(\int_{0}^{\infty}e^{2W\mathopen{}\mathclose{{\left(\frac{\rho^{2}_{1}}{4}s}}\right)-(\lambda_{1}+\Lambda)(\mu-1)s}ds<\frac{J^{1-\mu}(0)}{(\mu-1)\mathopen{}\mathclose{{\left(\frac{1}{2}\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{\mu}{\mu-p}}(x)dx}}\right)^{\frac{p-\mu}{p}}+\delta\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{n}{n-1}}(x)dx}}\right)^{1-n}}}\right)}\Bigg).

By performing the transformation
s\mapsto\frac{4t}{\rho^{2}_{1}},
we get

\displaystyle\mathbb{P}\mathopen{}\mathclose{{\left(\sigma^{\ast}_{1}=\infty}}\right)

\displaystyle=\mathbb{P}\Bigg(\frac{4}{\rho^{2}_{1}}\int_{0}^{\infty}e^{2W(t)-(\lambda_{1}+\Lambda)(\mu-1)\frac{4t}{\rho^{2}_{1}}}dt<\frac{J^{1-\mu}(0)}{(\mu-1)\mathopen{}\mathclose{{\left(\frac{1}{2}\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{\mu}{\mu-p}}(x)dx}}\right)^{\frac{p-\mu}{p}}+\delta\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{n}{n-1}}(x)dx}}\right)^{1-n}}}\right)}\Bigg)

\displaystyle=\mathbb{P}\Bigg(\frac{4}{\rho^{2}_{1}}\int_{0}^{\infty}e^{2(W(t)-\theta_{1}t)}dt<\frac{J^{1-\mu}(0)}{(\mu-1)\mathopen{}\mathclose{{\left(\frac{1}{2}\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{\mu}{\mu-p}}(x)dx}}\right)^{\frac{p-\mu}{p}}+\delta\mathopen{}\mathclose{{\left(\displaystyle\int_{D}\varphi^{\frac{n}{n-1}}(x)dx}}\right)^{1-n}}}\right)}\Bigg),

where
\theta_{1}=\displaystyle\frac{2(\lambda_{1}+\Lambda)(\mu-1)}{\rho^{2}_{1}}.
By Lemma
6.1
, we have

\displaystyle\mathbb{P}\mathopen{}\mathclose{{\left(\sigma^{\ast}_{1}=\infty}}\right)

\displaystyle=\mathbb{P}\Bigg(\frac{1}{X(\theta_{1},1)}<\frac{\rho^{2}_{1}J^{1-\mu}(0)}{2(\mu-1)\mathopen{}\mathclose{{\left(\frac{1}{2}\mathopen{}\mathclose{{\left(\int_{D}\varphi^{\frac{\mu}{\mu-p}}(x)dx}}\right)^{\frac{p-\mu}{p}}+\delta\mathopen{}\mathclose{{\left(\int_{D}\varphi^{\frac{n}{n-1}}(x)dx}}\right)^{1-n}}}\right)}\Bigg)

\displaystyle=\mathbb{P}\Bigg(2(\mu-1)\mathopen{}\mathclose{{\left(\frac{1}{2}\mathopen{}\mathclose{{\left(\int_{D}\varphi^{\frac{\mu}{\mu-p}}(x)dx}}\right)^{\frac{p-\mu}{p}}+\delta\mathopen{}\mathclose{{\left(\int_{D}\varphi^{\frac{n}{n-1}}(x)dx}}\right)^{1-n}}}\right)<X(\theta_{1},1)\rho^{2}_{1}J^{1-\mu}(0)\Bigg).

Case 2:
If
m+n>q>1,
by proceeding in the same way as in the case 2 of Theorem
5.1
and by using (
6.1
), we obtain

\displaystyle\sigma^{\ast}_{2}=\inf\Bigg\{t\geq 0:\int_{0}^{t}\mathopen{}\mathclose{{\left(e^{\eta(q-1)W(s)}\wedge e^{\eta(m+n-1)W(s)}}}\right)e^{-(\lambda_{1}+\Lambda)(q-1)s}ds\geq a_{1}\Bigg\}.

Note that
m+n-1>q-1>0
and so

\displaystyle\int_{0}^{t}\mathopen{}\mathclose{{\left(e^{\eta(q-1)W(s)}\wedge e^{\eta(m+n-1)W(s)}}}\right)e^{-(\lambda_{1}+\Lambda)(q-1)s}ds

\displaystyle=\int_{0}^{t}e^{\eta(m+n-1)W(s)-(\lambda+\Lambda)(q-1)s}\mathbf{1}_{\{W(s)<0\}}ds+\int_{0}^{t}e^{\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s}\mathbf{1}_{\{W(s)\geq 0\}}ds

\displaystyle\geq\int_{0}^{t}e^{\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s}\mathbf{1}_{\{W(s)\geq 0\}}ds,

and

\mathopen{}\mathclose{{\left\{\eta(q-1)W(t)-(\lambda_{1}+\Lambda)(q-1)t\geq 0}}\right\}\subseteq\mathopen{}\mathclose{{\left\{W(t)\geq 0}}\right\},

for all
t\geq 0
. We conclude that

\displaystyle\int_{0}^{t}\mathopen{}\mathclose{{\left(e^{\eta(q-1)W(s)}\wedge e^{\eta(m+n-1)W(s)}}}\right)e^{-(\lambda_{1}+\Lambda)(q-1)s}ds

\displaystyle\geq\int_{0}^{t}e^{-(\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s)}\mathbf{1}_{\{\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s\geq 0\}}ds.

Therefore, we define

\displaystyle\sigma^{\ast\ast}_{2}=\inf\Biggl\{t\geq 0:\int_{0}^{t}e^{-(\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s)}\mathbf{1}_{\{\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s\geq 0\}}ds\geq a_{1}\Biggr\}.

It can be easily seen that
\tau\leq\sigma^{\ast}_{2}\leq\sigma^{\ast\ast}_{2}.
From the definition of
\sigma^{\ast\ast}_{2}
and
[

, Theorem 2.6]
, we have

\displaystyle\mathbb{P}\{\tau<\infty\}

\displaystyle\geq\mathbb{P}\{\sigma^{\ast\ast}_{2}<\infty\}

\displaystyle=\mathbb{P}\Bigg(\int_{0}^{\infty}e^{-(\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s)}\mathbf{1}_{\{\eta(q-1)W(s)-(\lambda_{1}+\Lambda)(q-1)s\geq 0\}}ds\geq a_{1}\Bigg)

\displaystyle=\int^{\infty}_{a_{1}}(\lambda_{1}+\Lambda)(q-1)\sum_{n\geq 1}\exp\Bigg\{-\mathopen{}\mathclose{{\left(\frac{(\eta(q-1))^{2}}{8}j^{2}_{\frac{2(\lambda_{1}+\Lambda)(q-1)}{(\eta(q-1))^{2}}-1,n}}}\right)y\Bigg\}dy

\displaystyle=\frac{8(\lambda_{1}+\Lambda)(q-1)}{(\eta(q-1))^{2}}\sum_{n\geq 1}\frac{\exp\Bigg\{-\frac{(\eta(q-1))^{2}a_{1}}{8}j^{2}_{\frac{2(\lambda_{1}+\Lambda)(q-1)}{(\eta(q-1))^{2}}-1,n}\Bigg\}}{j^{2}_{\frac{2(\lambda_{1}+\Lambda)(q-1)}{(\eta(q-1))^{2}}-1,n}},

where we have used the Monotone Convergence Theorem to obtain the final equality, which completes the proof of the theorem.

The following theorem provides an upper bound for the probability of blow-up solution
v(\cdot,\cdot)
of the equation (

).

Theorem 6.3
.

Assume that
p,q,n>1,

m\geq 0
with
m+n\geq q
. If
f
is a non-negative bounded function and
|D|>k
, then an
upper bound for the probability of blow-up solution of the equation (

) is given by

\displaystyle\mathbb{P}\{\tau<\infty\}\leq\int_{\widetilde{N}_{1}}^{\infty}h_{3}(y)dy,

where

\displaystyle\widetilde{N}_{1}

\displaystyle=\frac{1}{2M(m+n-1)\|f\|_{\infty}^{m+n-1}}-\frac{1}{(\lambda+\Lambda)(q-1)},\ \mbox{here}\ M=\max\mathopen{}\mathclose{{\left\{|D|,\delta|D|}}\right\}\ \mbox{and}\

\displaystyle h_{3}(y)

\displaystyle=\frac{(2/(\eta(q-1))^{2}y)^{(2\Lambda(m+n-1)/(\eta(q-1))^{2})}}{y\Gamma(2\Lambda(m+n-1)/(\eta(q-1))^{2})}\exp\mathopen{}\mathclose{{\left(-\frac{2}{(\eta(q-1))^{2}y}}}\right).\

Proof.

By using the same procedure as in Theorem
6.1
, we have
\sigma_{\ast}\leq\tau,
where
\sigma_{\ast}
is given by

\displaystyle\sigma_{\ast}=\inf\Bigg\{t\geq 0:\int_{0}^{t}\mathopen{}\mathclose{{\left(e^{\eta\mathopen{}\mathclose{{\left(q-1}}\right)W(r)}\vee e^{\eta(m+n-1)W(r)}}}\right)e^{-\Lambda(m+n-1)r}dr\geq\frac{1}{2M(m+n-1)\|f\|_{\infty}^{m+n-1}}\Bigg\},

Note that
1<q\leq m+n
and so

\displaystyle\int_{0}^{t}\mathopen{}\mathclose{{\left(e^{\eta(q-1)W(s)}\wedge e^{\eta(m+n-1)W(s)}}}\right)e^{-\Lambda(m+n-1)s}ds

\displaystyle=\int_{0}^{t}e^{\eta(m+n-1)W(s)-\Lambda(m+n-1)s}\mathbf{1}_{\{W(s)<0\}}ds+\int_{0}^{t}e^{\eta(q-1)W(s)-\Lambda(m+n-1)s}\mathbf{1}_{\{W(s)\geq 0\}}ds

(6.4)

\displaystyle\sigma_{\ast\ast}=\inf\Bigg\{t\geq 0:\int_{0}^{t}e^{\eta(q-1)W(s)-\Lambda(m+n-1)s}ds

\displaystyle\geq\frac{1}{2M(m+n-1)\|f\|_{\infty}^{m+n-1}}-\frac{1}{\Lambda(m+n-1)}\Bigg\},

and
\sigma_{\ast\ast}\leq\sigma_{\ast}\leq\tau

\displaystyle\mathbb{P}\{\tau<\infty\}\leq\mathbb{P}(\sigma_{\ast\ast}<\infty)
