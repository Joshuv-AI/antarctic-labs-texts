Abstract

Theorem 1.1

Welch Bounds

Let

\displaystyle\sum_{1\leq j,k\leq n}|\langle\tau_{j},\tau_{k}\rangle|^{2m}=\sum_{j=1}^{n}\sum_{k=1}^{n}|\langle\tau_{j},\tau_{k}\rangle|^{2m}\geq\frac{n^{2}}{{d+m-1\choose m}},\quad\forall m\in\mathbb{N}.

\displaystyle\sum_{1\leq j,k\leq n}|\langle\tau_{j},\tau_{k}\rangle|^{2}=\sum_{j=1}^{n}\sum_{k=1}^{n}|\langle\tau_{j},\tau_{k}\rangle|^{2}\geq\frac{n^{2}}{{d}}.

\displaystyle\text{({Higher order Welch bounds})}\quad\max_{1\leq j,k\leq n,j\neq k}|\langle\tau_{j},\tau_{k}\rangle|^{2m}\geq\frac{1}{n-1}\left[\frac{n}{{d+m-1\choose m}}-1\right],\quad\forall m\in\mathbb{N}.

In particular,

\displaystyle\text{({First order Welch bound})}\quad\max_{1\leq j,k\leq n,j\neq k}|\langle\tau_{j},\tau_{k}\rangle|^{2}\geq\frac{n-d}{d(n-1)}.

Theorem 2.1
.

(First Order p-adic Welch Bound)
Let
p
be a prime and
n,d\in\mathbb{N}
. If
\{\tau_{j}\}_{j=1}^{n}
is any collection in
\mathbb{Q}_{p}^{d}

such that there exists
b\in\mathbb{Q}_{p}
satisfying

\displaystyle\sum_{j=1}^{n}\langle x,\tau_{j}\rangle\tau_{j}=bx,\quad\forall x\in\mathbb{Q}^{d}_{p},

then

\displaystyle\max_{1\leq j,k\leq n,j\neq k}\left\{\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2}\right|,|\langle\tau_{j},\tau_{k}\rangle|^{2}\right\}\geq\frac{1}{|d|}\left|\sum_{j=1}^{n}\langle\tau_{j},\tau_{j}\rangle\right|^{2}.

In particular, if
\langle\tau_{j},\tau_{j}\rangle=1
for all
1\leq j\leq n
, then

\displaystyle\text{{(First order p-adic Welch bound)}}\quad\max_{1\leq j,k\leq n,j\neq k}\{|n|,|\langle\tau_{j},\tau_{k}\rangle|^{2}\}\geq\frac{|n|^{2}}{|d|}.

Proof.

Define
S_{\tau}:\mathbb{Q}_{p}^{d}\ni x\mapsto\sum_{j=1}^{n}\langle x,\tau_{j}\rangle\tau_{j}\in\mathbb{Q}_{p}^{d}
. Then

\displaystyle bd=\operatorname{Tra}(bI_{\mathbb{Q}_{p}^{d}})=\operatorname{Tra}(S_{\tau})=\sum_{j=1}^{n}\langle\tau_{j},\tau_{j}\rangle,

\displaystyle b^{2}d=\operatorname{Tra}(b^{2}I_{\mathbb{Q}_{p}^{d}})=\operatorname{Tra}(S^{2}_{\tau})=\sum_{j=1}^{n}\sum_{k=1}^{n}\langle\tau_{j},\tau_{k}\rangle\langle\tau_{k},\tau_{j}\rangle.

Therefore

\displaystyle\left|\sum_{j=1}^{n}\langle\tau_{j},\tau_{j}\rangle\right|^{2}

\displaystyle=|\operatorname{Tra}(S_{\tau})|^{2}=|bd|^{2}=|d||b^{2}d|=|d|\left|\sum_{j=1}^{n}\sum_{k=1}^{n}\langle\tau_{j},\tau_{k}\rangle\langle\tau_{k},\tau_{j}\rangle\right|

\displaystyle=|d|\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2}+\sum_{j,k=1,j\neq k}^{n}\langle\tau_{j},\tau_{k}\rangle\langle\tau_{k},\tau_{j}\rangle\right|

\displaystyle\leq|d|\max_{1\leq j,k\leq n,j\neq k}\left\{\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2}\right|,|\langle\tau_{j},\tau_{k}\rangle\langle\tau_{k},\tau_{j}\rangle|\right\}

\displaystyle=|d|\max_{1\leq j,k\leq n,j\neq k}\left\{\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2}\right|,|\langle\tau_{j},\tau_{k}\rangle|^{2}\right\}.

Whenever
\langle\tau_{j},\tau_{j}\rangle=1
for all
1\leq j\leq n
,

\displaystyle|n|^{2}\leq|d|\max_{1\leq j,k\leq n,j\neq k}\{|n|,|\langle\tau_{j},\tau_{k}\rangle|^{2}\}.

∎

Theorem 2.3
.

(
Higher Order p-adic Welch Bounds
) Let
p
be a prime and
n,d,m\in\mathbb{N}
.
If
\{\tau_{j}\}_{j=1}^{n}
is any collection in
\mathbb{Q}_{p}^{d}

such that there exists
b\in\mathbb{Q}_{p}
satisfying

\displaystyle\sum_{j=1}^{n}\langle x,\tau_{j}^{\otimes m}\rangle\tau_{j}^{\otimes m}=bx,\quad\forall x\in\text{Sym}^{m}(\mathbb{Q}_{p}^{d}),

then

\displaystyle\max_{1\leq j,k\leq n,j\neq k}\left\{\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2m}\right|,|\langle\tau_{j},\tau_{k}\rangle|^{2m}\right\}\geq\frac{1}{\left|{d+m-1\choose m}\right|}\left|\sum_{j=1}^{n}\langle\tau_{j},\tau_{j}\rangle^{m}\right|^{2}.

In particular, if
\langle\tau_{j},\tau_{j}\rangle=1
for all
1\leq j\leq n
, then

\displaystyle\text{{(Higher order p-adic Welch bound)}}\quad\max_{1\leq j,k\leq n,j\neq k}\{|n|,|\langle\tau_{j},\tau_{k}\rangle|^{2m}\}\geq\frac{|n|^{2}}{\left|{d+m-1\choose m}\right|}.

Proof.

Define
S_{\tau}:\text{Sym}^{m}(\mathbb{Q}_{p}^{d})\ni x\mapsto\sum_{j=1}^{n}\langle x,\tau_{j}^{\otimes m}\rangle\tau_{j}^{\otimes m}\in\text{Sym}^{m}(\mathbb{Q}_{p}^{d})
. Then
Then

\displaystyle b\operatorname{dim(\text{Sym}^{m}(\mathbb{Q}_{p}^{d}))}=\operatorname{Tra}(bI_{\text{Sym}^{m}(\mathbb{Q}_{p}^{d})})=\operatorname{Tra}(S_{\tau})=\sum_{j=1}^{n}\langle\tau_{j}^{\otimes m},\tau_{j}^{\otimes m}\rangle,

\displaystyle b^{2}\operatorname{dim(\text{Sym}^{m}(\mathbb{Q}_{p}^{d}))}=\operatorname{Tra}(b^{2}I_{\text{Sym}^{m}(\mathbb{Q}_{p}^{d})})=\operatorname{Tra}(S^{2}_{\tau})=\sum_{j=1}^{n}\sum_{k=1}^{n}\langle\tau_{j}^{\otimes m},\tau^{\otimes m}_{k}\rangle\langle\tau^{\otimes m}_{k},\tau^{\otimes m}_{j}\rangle.

Therefore

\displaystyle\left|\sum_{j=1}^{n}\langle\tau_{j},\tau_{j}\rangle^{m}\right|^{2}=\left|\sum_{j=1}^{n}\langle\tau_{j}^{\otimes m},\tau_{j}^{\otimes m}\rangle\right|^{2}=|\operatorname{Tra}(S_{\tau})|^{2}=\left|b\operatorname{dim(\text{Sym}^{m}(\mathbb{Q}_{p}^{d}))}\right|^{2}

\displaystyle=\left|\operatorname{dim(\text{Sym}^{m}(\mathbb{Q}_{p}^{d}))}\right|\left|b^{2}\operatorname{dim(\text{Sym}^{m}(\mathbb{Q}_{p}^{d}))}\right|

\displaystyle=\left|\operatorname{dim(\text{Sym}^{m}(\mathbb{Q}_{p}^{d}))}\right|\left|\sum_{j=1}^{n}\sum_{k=1}^{n}\langle\tau_{j}^{\otimes m},\tau_{k}^{\otimes m}\rangle\langle\tau_{k}^{\otimes m},\tau_{j}^{\otimes m}\rangle\right|

\displaystyle=\left|{d+m-1\choose m}\right|\left|\sum_{j=1}^{n}\sum_{k=1}^{n}\langle\tau_{j}^{\otimes m},\tau_{k}^{\otimes m}\rangle\langle\tau_{k}^{\otimes m},\tau_{j}^{\otimes m}\rangle\right|

\displaystyle=\left|{d+m-1\choose m}\right|\left|\sum_{j=1}^{n}\sum_{k=1}^{n}\langle\tau_{j},\tau_{k}\rangle^{m}\langle\tau_{k},\tau_{j}\rangle^{m}\right|

\displaystyle=\left|{d+m-1\choose m}\right|\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2m}+\sum_{j,k=1,j\neq k}^{n}\langle\tau_{j},\tau_{k}\rangle^{m}\langle\tau_{k},\tau_{j}\rangle^{m}\right|

\displaystyle\leq\left|{d+m-1\choose m}\right|\max_{1\leq j,k\leq n,j\neq k}\left\{\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2m}\right|,|\langle\tau_{j},\tau_{k}\rangle^{m}\langle\tau_{k},\tau_{j}\rangle^{m}|\right\}

\displaystyle=\left|{d+m-1\choose m}\right|\max_{1\leq j,k\leq n,j\neq k}\left\{\left|\sum_{l=1}^{n}\langle\tau_{l},\tau_{l}\rangle^{2m}\right|,|\langle\tau_{j},\tau_{k}\rangle|^{2m}\right\}.

Whenever
\langle\tau_{j},\tau_{j}\rangle=1
for all
1\leq j\leq n
,

\displaystyle|n|^{2}\leq\left|{d+m-1\choose m}\right|\max_{1\leq j,k\leq n,j\neq k}\{|n|,|\langle\tau_{j},\tau_{k}\rangle|^{2m}\}.

∎
