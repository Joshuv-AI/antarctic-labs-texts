\begin{split}\lambda_{1}>\left(\frac{2-p}{p}+\frac{2}{p(p-1)}-\bigg(\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\bigg)\right)^{-1}\left(\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\right)\end{split}

Then,

\operatorname{dim}\mathcal{H}^{1}\left(L^{p}(M)\right)<\infty,

where
\mathcal{H}^{1}\left(L^{p}(M)\right)
denotes the space of
L^{p}
harmonic 1-forms on
M
.

Proof.

Recall the inequality we have proved

\begin{split}|\omega|^{p-1}\Delta|\omega|^{p-1}\geq&A_{p,n}|\nabla|\omega|^{p-1}|^{2}-\bigg\langle\delta d(|\omega|^{p-2}\omega),|\omega|^{p-2}\omega\bigg\rangle\\
&+|\omega|^{2p-4}\left(-k^{2}-\frac{n-1}{n}S\right)|\omega|^{2}.\end{split}

(4.19)

As in (
4.2
), we have

\begin{split}&|\omega|^{\frac{p}{2}}\Delta|\omega|^{\frac{p}{2}}\\
\geq&\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}\right)|\nabla|\omega|^{\frac{p}{2}}|^{2}-\frac{p}{2p-2}\bigg\langle\delta d(|\omega|^{p-2}\omega),\omega\bigg\rangle\\
&+\frac{p}{2p-2}\left(-k^{2}-\frac{n-1}{n}S\right)|\omega|^{p}.\end{split}

(4.20)

Multiplying both sides by
\phi^{2}

\begin{split}&\int_{M}\phi^{2}\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}\right)|\nabla|\omega|^{\frac{p}{2}}|^{2}-\frac{p}{2p-2}\int_{M}\phi^{2}\bigg\langle\delta d(|\omega|^{p-2}\omega),\omega\bigg\rangle\\
&+\int_{M}\phi^{2}\frac{p}{2p-2}\left(-k^{2}-\frac{n-1}{n}S\right)|\omega|^{p}\\
\leq&\int_{M}\phi^{2}|\omega|^{\frac{p}{2}}\Delta|\omega|^{\frac{p}{2}}\\
=&\int_{M}|\nabla\phi|^{2}|\omega|^{p}-\int_{M}|\nabla(\phi|\omega|^{p})|^{2}.\end{split}

(4.21)

However, by
-k^{2}\leq\overline{\operatorname{BiRic}}^{a}
, we know

\begin{split}&\int_{M\backslash B_{R}(o)}(-\frac{k^{2}}{a+1}+S)\varphi^{2}|\omega|^{p}\\
\leq&\int_{M\backslash B_{R}(o)}(\overline{\operatorname{Ric}}(N)+S)\varphi^{2}|\omega|^{p}\\
\leq&\int_{M\backslash B_{R}(o)}\left|\nabla\left(\varphi|\omega|^{\frac{p}{2}}\right)\right|^{2}.\end{split}

(4.22)

Thus, we get

\begin{split}&\int_{M\backslash B_{R}(o)}(S\varphi^{2}|\omega|^{p}\leq\int_{M\backslash B_{R}(o)}\left|\nabla\left(\varphi|\omega|^{\frac{p}{2}}\right)\right|^{2}+\frac{k^{2}}{a+1}\int_{M\backslash B_{R}(o)}\varphi^{2}|\omega|^{p}.\end{split}

(4.23)

Combining (
4.21
) and (
4.23
), we have

\begin{split}&\int_{M}\phi^{2}\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}\right)|\nabla|\omega|^{\frac{p}{2}}|^{2}\\
&-\frac{p}{2p-2}\int_{M}\phi^{2}\bigg\langle\delta d(|\omega|^{p-2}\omega),\omega\bigg\rangle-\int_{M}\frac{p}{2p-2}k^{2}\phi^{2}|\omega|^{p}\\
\leq&\int_{M}|\nabla\phi|^{2}|\omega|^{p}-\int_{M}|\nabla(\phi|\omega|^{p})|^{2}\\
&+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\bigg[\int_{M\backslash B_{R}(o)}\left|\nabla\left(\varphi|\omega|^{\frac{p}{2}}\right)\right|^{2}+\frac{k^{2}}{a+1}\int_{M\backslash B_{R}(o)}\varphi^{2}|\omega|^{p}\bigg],\end{split}

(4.24)

which can be written as

\begin{split}&\int_{M}\phi^{2}\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}\right)|\nabla|\omega|^{\frac{p}{2}}|^{2}\\
&-\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\int_{M}\phi^{2}|\omega|^{p}-\int_{M}|\nabla\phi|^{2}|\omega|^{p}\\
\leq&\bigg[\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\bigg]\int_{M\backslash B_{R}(o)}\left|\nabla\left(\varphi|\omega|^{\frac{p}{2}}\right)\right|^{2}+\frac{2p}{p-1}\frac{p-2}{p}\int_{M}\Big|\nabla|\omega|^{\frac{p}{2}}\Big|\phi|\nabla\phi||\omega|^{\frac{p}{2}}.\end{split}

(4.25)

Thus, we have

\begin{split}&\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}\right)\int_{M}\phi^{2}|\nabla|\omega|^{\frac{p}{2}}|^{2}\\
&-\int_{M}|\nabla\phi|^{2}|\omega|^{p}\\
\leq&\bigg[\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}\bigg]\int_{M\backslash B_{R}(o)}\left|\nabla\left(\varphi|\omega|^{\frac{p}{2}}\right)\right|^{2}\\
&+2\frac{p-2}{p-1}\frac{1}{4\epsilon_{1}}\int_{M}|\nabla\phi|^{2}|\omega|^{p}.\end{split}

(4.26)

Following
[
12
]
, if
\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}\leq 0,
then by (
4.26
), we have

\begin{split}&\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}\right)\int_{M}\phi^{2}|\nabla|\omega|^{\frac{p}{2}}|^{2}\\
\leq&2\frac{p-2}{p-1}\frac{1}{4\epsilon_{1}}\int_{M}|\nabla\phi|^{2}|\omega|^{p}.\end{split}

By the conditions on
p
and (
4.10
)(
4.11
)(
4.12
), we have

\begin{split}\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}>0.\end{split}

If
\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}>0,
then by (
4.26
), we have

\begin{split}&\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}\right)\int_{M}\phi^{2}|\nabla|\omega|^{\frac{p}{2}}|^{2}\\
\leq&\left(1+2\frac{p-2}{p-1}\frac{1}{4\epsilon_{1}}\right)\int_{M}|\nabla\phi|^{2}|\omega|^{p}\\
&+\bigg[\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}\bigg]\\
&\bigg((1+\frac{1}{\epsilon_{2}})\int_{M}|\nabla\phi|^{2}|\omega|^{p}+(1+\epsilon_{2})\int_{M}|\phi|^{2}|\nabla|\omega|^{\frac{p}{2}}|^{2}\bigg).\end{split}

(4.27)

It follows that

\begin{split}&\left(\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}-B(1+\epsilon_{2})\right)\int_{M}\phi^{2}|\nabla|\omega|^{\frac{p}{2}}|^{2}\\
\leq&\left(1+2\frac{p-2}{p-1}\frac{1}{4\epsilon_{1}}\right)\int_{M}|\nabla\phi|^{2}|\omega|^{p}+B(1+\frac{1}{\epsilon_{2}})\int_{M}|\nabla\phi|^{2}|\omega|^{p},\end{split}

(4.28)

where
B=\bigg[\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}\bigg].

Note it suffices to prove that we can choose
\epsilon_{1},\epsilon_{2}
such that

\begin{split}&\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}-B(1+\epsilon_{2})\\
=&\frac{2-p}{p}+\frac{1}{2p-2}\frac{4}{p}-\bigg\{\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\\
&+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}\bigg\}-\epsilon_{1}-B\epsilon_{2}>0.\end{split}

If
p>1+\sqrt{n-1},

\begin{split}\frac{2-p}{p}+\frac{1}{2p-2}\frac{4}{p}-\bigg(\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\bigg)>0.\end{split}

which holds if
p<\frac{4n}{n-1}.
Moreover,

\begin{split}\lambda_{1}>&\left(\frac{2-p}{p}+\frac{1}{2p-2}\frac{4}{p}-\bigg(\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\bigg)\right)^{-1}\\
&\times\left(\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\right).\end{split}

Thus in the case where
p>1+\sqrt{n-1},
we can choose
\epsilon_{1},\epsilon_{2}
such that

\begin{split}&\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}-A(1+\epsilon_{2})>0.\end{split}

If
p\leq 1+\sqrt{n-1},
then

\begin{split}&\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}-A(1+\epsilon_{2})\\
=&\frac{2-p}{p}+\frac{1}{2p-2}\frac{(p-1)^{2}}{(m-1)}\frac{4}{p}\\
&-\bigg\{\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1+\bigg[\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\bigg]\frac{1}{\lambda_{1}(M)}\bigg\}-\epsilon_{1}-A\epsilon_{2}.\end{split}

In this case, if

\begin{split}\frac{2-p}{p}+\frac{1}{2p-2}\frac{(p-1)^{2}}{(n-1)}\frac{4}{p}-\bigg(\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\bigg)>0.\end{split}

which holds if
p<\frac{4n}{n-1}
. In additon,

\begin{split}\lambda_{1}>&\left(\frac{2-p}{p}+\frac{2(p-1)}{p(n-1)}-\bigg(\frac{p}{2p-2}\left(\frac{n-1}{n}\right)-1\bigg)\right)^{-1}\\
&\times\left(\frac{p}{2p-2}k^{2}+\frac{p}{2p-2}\left(\frac{n-1}{n}\right)\frac{k^{2}}{a+1}\right),\end{split}

Thus in the case where
p\leq 1+\sqrt{n-1},
we can choose
\epsilon_{1},\epsilon_{2}
such that

\begin{split}&\frac{2-p}{p}+\frac{p}{2p-2}A_{p,n}\frac{4(p-1)^{2}}{p^{2}}-\epsilon_{1}-A(1+\epsilon_{2})>0.\end{split}

Hence, from (
4.28
), we get

\displaystyle\int_{M}|\nabla|\omega|^{\frac{p}{2}}|^{2}\phi^{2}\leq C\int_{M}|\nabla\phi|^{2}|\omega|^{p}.

(4.29)

The remained proof is the same as that in Theorem
4.1
.
