def sq2(n):
c=0
for i in range(2^n):
l=list(Integer(i).digits(base=2,padto=n))
if l[-1]==1:
cc=1/4
for j in range(n-2):
ll=[l[j],l[j+1],l[j+2]]
if ll==[0,0,0]:
cc*=1/2
if ll==[0,0,1]:
cc*=1/2
if ll==[0,1,0]:
cc*=(1-sqrt(2)/2)
if ll==[0,1,1]:
cc*=sqrt(2)/2
if ll==[1,0,0]:
cc*=1
if ll==[1,0,1]:
cc=0
break
if ll==[1,1,0]:
cc*=(1-sqrt(2)/2)
if ll==[1,1,1]:
cc*=sqrt(2)/2
c+=cc
return [c.expand(),c.n()]
