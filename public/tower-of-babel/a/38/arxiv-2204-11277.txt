Lemma 2.5
.

Let
m\geq 2
,
s\leq 0
,
\sigma\geq d/2-2/(m-1)
. Suppose that
\mathrm{supp}\mathscr{F}_{x}u_{j}(t)\subset\mathbb{R}^{d}_{I},~1\leq j\leq m,\ t\geq 0
. Then we have, for some
C_{m}>0
,

\displaystyle\|u_{1}u_{2}\cdots u_{m}\|_{\widetilde{L}^{1}(0,\infty;E^{s,\sigma}_{2,2})}\leq C_{m}\prod^{m}_{j=1}\|u_{j}\|_{\widetilde{L}^{m}(0,\infty;E^{s,\sigma+2/m}_{2,2})},

(27)

where
C_{m}\leq C^{m}m^{m/2}
if
\sigma\geq d/2
,
C
depends only on
s,d,\sigma
.

Proof.

By definition and the support set property of
\mathscr{F}u_{j}(t)
, we have

\displaystyle\|u_{1}u_{2}\cdots u_{m}\|_{\widetilde{L}^{1}(0,\infty;E^{s,\sigma}_{2,2})}=\left(\sum_{k\in\mathbb{Z}^{d}_{I}}\langle k\rangle^{2\sigma}2^{2s|k|}\|\Box_{k}(u_{1}u_{2}...u_{m})\|^{2}_{L^{1}_{t}L^{2}_{x}(\mathbb{R}_{+}\times\mathbb{R}^{d})}\right)^{1/2}.

(28)

Using the definition of
\Box_{k}
, we see that

\displaystyle\|\Box_{k}(u_{1}u_{2}\cdots u_{m})\|_{L_{t}^{1}L_{x}^{2}}

\displaystyle\leq\sum_{k_{1},...,k_{m}\in\mathbb{Z}^{d}_{I}}\|\Box_{k}(\Box_{k_{1}}u_{1}\Box_{k_{2}}u_{2}\cdots\Box_{k_{m}}u_{m})\|_{L_{t}^{1}L_{x}^{2}}.

By the orthogonality of
\Box_{k}
,

\Box_{k}(\Box_{k_{1}}u_{1}\Box_{k_{2}}u_{2}\cdots\Box_{k_{m}}u_{m})\neq 0,

implies that
(k_{1},...,k_{m})
belongs to

\Lambda_{k}(k_{1},...,k_{m}):=\left\{(k_{1},...,k_{m}):\,-1\leq k(j)-k_{1}(j)-...-k_{m}(j)\leq m,\ \ j=1,...,d\right\}.

It follows that

\displaystyle\|\Box_{k}(u_{1}u_{2}\cdots u_{m})\|_{L_{t}^{1}L_{x}^{2}}

\displaystyle\leq\sum_{k_{1},...,k_{m}\in\mathbb{Z}^{d}_{I}}\|\Box_{k_{1}}u_{1}\Box_{k_{2}}u_{2}\cdots\Box_{k_{m}}u_{m}\|_{L_{t}^{1}L_{x}^{2}}\chi_{\Lambda_{k}(k_{1},...,k_{m})}.

(29)

Let us further write

(\mathbb{Z}^{d}_{I})^{m,i}:=\left\{(k_{1},...,k_{m})\in(\mathbb{Z}^{d}_{I})^{m}:\,|k_{i}|=\max_{1\leq j\leq m}|k_{j}|\right\}.

One has that

\displaystyle\|\Box_{k}(u_{1}u_{2}\cdots u_{m})\|_{L_{t}^{1}L_{x}^{2}}

\displaystyle\leq\sum^{m}_{i=1}\sum_{(k_{1},...,k_{m})\in(\mathbb{Z}^{d}_{I})^{m,i}}\|\Box_{k_{1}}u_{1}\Box_{k_{2}}u_{2}\cdots\Box_{k_{m}}u_{m}\|_{L_{t}^{1}L_{x}^{2}}\chi_{\Lambda_{k}(k_{1},...,k_{m})}

\displaystyle:=\sum^{m}_{i=1}L_{i}.

(30)

By symmetry, it suffices to estimate
L_{m}
. By Hölder’s inequality and
\|\Box_{k}u\|_{\infty}\leq\|\Box_{k}u\|_{2}
,

\displaystyle L_{m}

\displaystyle\leq\sum_{(k_{1},...,k_{m})\in(\mathbb{Z}^{d}_{I})^{m,m}}\prod^{m}_{j=1}\|\Box_{k_{j}}u_{j}\|_{L^{m}_{t}L^{\infty}_{x}}\|\Box_{k_{m}}u_{m}\|_{L_{t}^{m}L_{x}^{2}}\chi_{\Lambda_{k}(k_{1},...,k_{m})}

\displaystyle\leq\sum_{(k_{1},...,k_{m})\in(\mathbb{Z}^{d}_{I})^{m,m}}\prod^{m}_{j=1}\|\Box_{k_{j}}u_{j}\|_{L^{m}_{t}L^{2}_{x}}\chi_{\Lambda_{k}(k_{1},...,k_{m})}.

(31)

Let us write

(\mathbb{Z}^{d}_{I})^{m-1}_{r}:=\left\{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}:\,|k_{1}|,...,|k_{m-1}|\leq r\right\}.

Then for
l_{m}=k-k_{1}-...-k_{m-1}
,
\mathbb{A}=\{l\in\mathbb{Z}^{d}:\ -1\leq l(j)\leq m,\,j=1,...,d\}
,

\displaystyle L_{m}

\displaystyle\leq\sum_{l\in\mathbb{A}}\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|l_{m}-l|}}\prod^{m-1}_{j=1}\|\Box_{k_{j}}u_{j}\|_{L^{m}_{t}L^{2}_{x}}\|\Box_{l_{m}-l}u_{m}\|_{L^{m}_{t}L^{2}_{x}}\chi_{\{l_{m}-l\in\mathbb{Z}^{d}_{I}\}}.

(32)

Using the Cauchy-Schwarz inequality, one has that

\displaystyle L_{m}\leq

\displaystyle\sum_{l\in\mathbb{A}}\left(\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|l_{m}-l|}}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{2(\sigma+2/m)}2^{2s|k_{j}|}\|\Box_{k_{j}}u_{j}\|^{2}_{L^{m}_{t}L^{2}_{x}}\right)^{1/2}

\displaystyle\times\left(\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|l_{m}-l|}}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{-2(\sigma+2/m)}2^{-2s|k_{j}|}\|\Box_{l_{m}-l}u_{m}\|^{2}_{L_{t}^{m}L_{x}^{2}}\chi_{\{l_{m}-l\in\mathbb{Z}^{d}_{I}\}}\right)^{1/2}

\displaystyle\lesssim

\displaystyle\ 2^{|s|dm}2^{-s|k|}\prod^{m-1}_{j=1}\|u_{j}\|_{\widetilde{L}^{m}(0,\infty;E^{s,\sigma+2/m}_{2,2})}

\displaystyle\times\sum_{l\in\mathbb{A}}\left(\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|l_{m}-l|}}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{-2(\sigma+2/m)}\left(2^{s|l_{m}-l|}\|\Box_{l_{m}-l}u_{m}\|_{L_{t}^{m}L_{x}^{2}}\right)^{2}\right)^{1/2}.

(33)

By (
33
), we have for
l_{m}=k-k_{1}-...-k_{m-1}
,

\displaystyle\left(\sum_{k\in\mathbb{Z}^{d}_{I}}\langle k\rangle^{2\sigma}2^{2s|k|}L_{m}^{2}\right)^{1/2}

\displaystyle\lesssim\ 2^{|s|dm}\sum_{l\in\mathbb{A}}\left(\sum_{k\in\mathbb{Z}^{d}_{I}}\langle k\rangle^{2\sigma}\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|l_{m}-l|}}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{-2(\sigma+2/m)}2^{2s|l_{m}-l|}\|\Box_{l_{m}-l}u_{m}\|_{L_{t}^{m}L_{x}^{2}}^{2}\right)^{1/2}

\displaystyle\hskip 10.00002pt\hskip 10.00002pt\times\prod^{m-1}_{j=1}\|u_{j}\|_{\widetilde{L}^{m}(0,\infty;E^{s,\sigma+2/m}_{2,2})}.

(34)

Noticing that
|l|\leq dm
for
l\in\mathbb{A}
, we have

\displaystyle\sum_{k\in\mathbb{Z}^{d}_{I}}

\displaystyle\langle k\rangle^{2\sigma}\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|l_{m}-l|}}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{-2(\sigma+2/m)}2^{2s|l_{m}-l|}\|\Box_{l_{m}-l}u_{m}\|_{L_{t}^{m}L_{x}^{2}}^{2}

\displaystyle\leq\sum_{k_{m}\in\mathbb{Z}^{d}_{I}}\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|k_{m}|}}\left\langle\sum^{m}_{j=1}k_{j}+l\right\rangle^{2\sigma}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{-2(\sigma+2/m)}\left(2^{s|k_{m}|}\|\Box_{k_{m}}u_{m}\|_{L_{t}^{m}L_{x}^{2}}\right)^{2}

\displaystyle\lesssim m^{2|\sigma|}\sum_{k_{m}\in\mathbb{Z}^{d}_{I}}\sum_{(k_{1},...,k_{m-1})\in(\mathbb{Z}^{d}_{I})^{m-1}_{|k_{m}|}}\prod^{m-1}_{j=1}\langle k_{j}\rangle^{-2(\sigma+2/m)}\langle k_{m}\rangle^{-4/m}

\displaystyle\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\hskip 10.00002pt\times\left(\langle k_{m}\rangle^{\sigma+2/m}2^{s|k_{m}|}\|\Box_{k_{m}}u_{m}\|_{L_{t}^{m}L_{x}^{2}}\right)^{2}.

If

u_{t}-\Delta u=e^{u}-1,\ \ u(0,x)=u_{0}(x),

(87)

where
u(t,x)
is a complex-valued function of
(t,x)\in[0,\infty)\times\mathbb{R}^{d}
,
u_{0}\in{E}^{s}_{\sigma}
with
s<0
and
\widehat{u}_{0}
is supported in the first octant and away from the unit ball, say

\displaystyle{\rm supp}\,\widehat{u}_{0}\subset\{\xi\in\mathbb{R}^{d}:\ \xi_{i}\geq 0,\ i=1,...,d,\ |\xi|_{\infty}\geq 2\}.

(88)

The main result of this section is the following

Theorem 4.1
.

Let
s<0
,
\sigma\geq d/2
, and
u_{0}\in E^{s}_{\sigma}
satisfy (
88
) Then there exists
s_{0}\leq s
such that (
87
) has a unique solution

u\in C(\mathbb{R}_{+},E^{s_{0}}_{\sigma})\cap\widetilde{L}^{\infty}(0,\infty;E^{s_{0},\sigma}_{2,2})\cap\widetilde{L}^{1}(0,\infty;E^{s_{0},\sigma+2}_{2,2})

in the sense that
u
satisfies the equivalent integral equation

\displaystyle u(t)=e^{t(I+\Delta)}u_{0}+\int^{t}_{0}e^{(t-\tau)(I+\Delta)}(e^{u(\tau)}-u(\tau)-1)d\tau.

(89)

Recalling that

\displaystyle e^{u}=\sum^{\infty}_{m=0}\frac{u^{m}}{m!},

(90)

we see that Fujita’s critical and subcritical powers are contained in the Taylor expansion of
e^{u}
in the cases
m=2,3
for
d=1
, and
m=2
for
d=2
. Moreover, the linear part
u
contained in the Taylor’s expansion of
e^{u}
plays a bad role which prevents the global existence of solutions. Indeed, let us observe the semi-group
e^{t(I+\Delta)}
, its lower frequency part has no time-decay and one easily sees that

e^{t/2}\|P_{\leq 1/2}u_{0}\|_{2}\leq\|e^{t(I+\Delta)}P_{\leq 1/2}u_{0}\|_{2}\leq e^{t}\|u_{0}\|_{2},\ \ P_{\leq 1/2}=\mathscr{F}^{-1}\chi_{|\xi|\leq 1/2}\mathscr{F}.

So, the lower frequency part of the free solutions tends to infinity as
t\to\infty
. This is why we assume that the support set of the Fourier transform of initial data is away from the unit ball in (
88
).

4.1
Linear Estimates

Now, let us make a scaling to (
87
). Denote

u_{\lambda}(t,x)=u(\lambda^{2}t,\lambda x).

If
u
is a solution of (
87
), we easily see that
u_{\lambda}
solves

u_{t}-\Delta u=\lambda^{2}(e^{u}-1),\ \ u(0,x)=u_{0}(\lambda x).

(91)

(
91
) can be rewritten as

u_{t}-(\lambda^{2}+\Delta)u-\lambda^{2}(e^{u}-u-1)=0,\ \ u(0,x)=u_{0}(\lambda x),

which is essentially equivalent to the following integral equation

\displaystyle u(t)=e^{t(\lambda^{2}I+\Delta)}u_{0}(\lambda\,\cdot)+\lambda^{2}\int^{t}_{0}e^{(t-\tau)(\lambda^{2}I+\Delta)}(e^{u(\tau)}-u(\tau)-1)d\tau.

(92)

We can first solve (
92
). For the sake of convenience, we write

H_{\lambda}(t):=e^{t(\lambda^{2}I+\Delta)}=\mathscr{F}^{-1}e^{-t(|\xi|^{2}-\lambda^{2})}\mathscr{F}.

Let us consider the estimate of
H_{\lambda}(t)
. As indicated in the above,
H_{\lambda}(t)
has no time-decay for
|\xi|\leq\lambda
.

Lemma 4.2
.

Let
1\leq p\leq\infty
,
\lambda\gg 1
. Assume that
k\in\mathbb{Z}^{d}
and
|k|_{\infty}\geq 2\lambda
. Then we have

\displaystyle\|\Box_{k}H_{\lambda}(t)u_{0}\|_{p}\leq Ce^{-ct|k|^{2}}\|\Box_{k}u_{0}\|_{p},

(93)

where
C
is independent of
\lambda
and
k
.

Proof
. Let
\sigma
be a smooth cut-off function verifying
\sigma(\xi)=1
for
\xi\in[0,1]^{d}
and
\sigma(\xi)=0
if
\xi\not\in[-1/4,5/4]^{d}
. Put
\sigma_{k}=\sigma(\cdot-\xi)
. By Young’s inequality,

\|\Box_{k}H_{\lambda}(t)u_{0}\|_{p}\leq\left\|\mathscr{F}^{-1}\left(\sigma_{k}e^{t(\lambda^{2}-|\xi|^{2})}\right)\right\|_{1}\|\Box_{k}u_{0}\|_{p}.

Using the translation invariance,

\left\|\mathscr{F}^{-1}\left(\sigma_{k}e^{t(\lambda^{2}-|\xi|^{2})}\right)\right\|_{1}=e^{t(\lambda^{2}-|k|^{2})}\left\|\mathscr{F}^{-1}\left(\sigma e^{-t(|\xi|^{2}+2k\xi)}\right)\right\|_{1}.

Noticing that
{\rm supp}\,\sigma
is compact, in view of Proposition
1.3
we have

\left\|\mathscr{F}^{-1}\left(\sigma e^{-t(|\xi|^{2}+2k\xi)}\right)\right\|_{1}\lesssim e^{Ct|k|}.

Noticing that
|k|_{\infty}\geq 2\lambda
, one has the result as desired.
\hfill\Box

Lemma 4.3
.

Let
1\leq\gamma,\gamma_{1},p\leq\infty
,
\gamma_{1}\leq\gamma
,
\lambda\gg 1
. Assume that

\displaystyle{\rm supp}\,\widehat{u}_{0},{\rm supp}\,\widehat{f(t)}\subset\mathbb{R}^{d}_{I,\lambda}:=\{\xi\in\mathbb{R}^{d}_{I}:|\xi|_{\infty}\geq 2\lambda\}.

(94)

Then we have

\displaystyle\|\Box_{k}H_{\lambda}(t)u_{0}\|_{L^{\gamma}_{t}L^{p}_{x}}

\displaystyle\leq C|k|^{-2/\gamma}\|\Box_{k}u_{0}\|_{p},

(95)

\displaystyle\left\|\int^{t}_{0}\Box_{k}H_{\lambda}(t-\tau)f(\tau)d\tau\right\|_{L^{\gamma}_{t}L^{p}_{x}}

\displaystyle\leq C|k|^{-2/\gamma-2/\gamma^{\prime}_{1}}\left\|\Box_{k}f\right\|_{L^{\gamma^{\prime}_{1}}_{t}L^{p}_{x}},

(96)

where
C
is independent of
\lambda
and
k
.

Proof
. Taking the
L^{\gamma}
norm in both sides of (
93
), we immediately have (
95
). By (
93
),

\displaystyle\left\|\int^{t}_{0}\Box_{k}H_{\lambda}(t-\tau)f(\tau)d\tau\right\|_{p}\leq\int^{t}_{0}e^{-c(t-\tau)|k|^{2}}\|\Box_{k}f(\tau)\|_{p}d\tau.

(97)

Taking
L^{\gamma}
norm in both sides of (
97
) and using Young’s inequality, we obtain (
96
).
\hfill\Box

Corollary 4.4
.

Let
s\leq 0
,
\sigma\in\mathbb{R}
,
1\leq\gamma_{1}\leq\gamma\leq\infty
. Suppose that
\mathrm{supp}~\mathscr{F}_{x}f,\,{\rm supp}\,\widehat{u}_{0}\subset\mathbb{R}^{d}_{I,\lambda}
. Then there exists
C>0
which is independent of
\lambda\gg 1
such that

\displaystyle\|H_{\lambda}(t)u_{0}\|_{\widetilde{L}^{\gamma}(0,\infty;E^{s,\sigma+2/\gamma}_{2,2})}

\displaystyle\leq C\|u_{0}\|_{E^{s}_{\sigma}},

(98)

\displaystyle\left\|\int_{0}^{t}H_{\lambda}(t-\tau)f(\tau)~d\tau\right\|_{\widetilde{L}^{\gamma}(0,\infty;E^{s,\sigma+2/\gamma}_{2,2})}

\displaystyle\leq C\|f\|_{\widetilde{L}^{\gamma_{1}}(0,\infty;E^{s,\sigma-2/\gamma^{\prime}_{1}}_{2,2})}.

(99)

Proof.

Taking
p=2
, multiplying
2^{s|k|}\langle k\rangle^{\sigma+2/\gamma}
and then equipping
\ell^{2}
norms in both sides of (
95
) and (
96
), we have the results, as desired.
∎
