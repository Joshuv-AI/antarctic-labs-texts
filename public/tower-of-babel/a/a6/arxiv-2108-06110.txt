\displaystyle\mathcal{H}_{\mathrm{eff}}=\frac{G_{F}}{\sqrt{2}}V_{cs}\Big(\bar{s}\gamma^{\mu}(1-\gamma_{5})c\Big)\Big(\bar{\nu}_{\ell}\gamma_{\mu}(1-\gamma_{5})\ell\Big),

,
\Gamma(\Xi_{c}\rightarrow\Lambda\pi^{-}\mu^{+}\nu_{\nu})=1.037(131)\times 10^{-13}s^{-1}
,
\Gamma(\Omega_{c}\rightarrow\Lambda K^{-}e^{+}\nu_{e})=8.889(344)\times 10^{-15}s^{-1}
,
\Gamma(\Omega_{c}\rightarrow\Lambda K^{-}\mu^{+}\nu_{\nu})=8.771(343)\times 10^{-15}s^{-1}
, and
\mathcal{B}(\Lambda_{c}^{+}\rightarrow p\pi^{-}e^{+}\nu_{e})=2.48(15)\%
,
\mathcal{B}(\Lambda_{c}^{+}\rightarrow p\pi^{-}\mu^{+}\nu_{\mu})=2.50(14)\%

and so on. This work can provide a theoretical basis for the ongoing experiments at BESIII, LHCb and BELLE-II.

In this part, we will introduce the spinors and polarization vectors of fermions and vector boson
Auvil:1966eao
. The spinors for spin-1/2 fermions are

\displaystyle u\left(\vec{p},+\frac{1}{2}\right)=

\displaystyle N_{p}\left(\begin{array}[]{c}e^{-i\phi/2}\cos\left(\theta/2\right)\\
e^{i\phi/2}\sin\left(\theta/2\right)\\
\frac{\left|\vec{p}\right|}{E+m}e^{-i\phi/2}\cos\left(\theta/2\right)\\
\frac{\left|\vec{p}\right|}{E+m}e^{i\phi/2}\sin\left(\theta/2\right)\end{array}\right),

\displaystyle u\left(\vec{p},-\frac{1}{2}\right)=

\displaystyle N_{p}\left(\begin{array}[]{c}-e^{-i\phi/2}\sin\left(\theta/2\right)\\
e^{i\phi/2}\cos\left(\theta/2\right)\\
\frac{\left|\vec{p}\right|}{E+m}e^{-i\phi/2}\sin\left(\theta/2\right)\\
-\frac{\left|\vec{p}\right|}{E+m}e^{i\phi/2}\cos\left(\theta/2\right)\end{array}\right),

where
N_{p}=\sqrt{E+m}
and
p^{\mu}=(p^{0},\vec{p}\sin\theta\cos\phi,\vec{p}\sin\theta\sin\phi,\vec{p}\cos\theta)

\bar{v}(\vec{p},s)=-i\gamma^{2}u^{\star}(\vec{p},s)
. The polarization vectors can be expressed as

\displaystyle\epsilon^{\mu}(+1)=

\displaystyle\frac{1}{\sqrt{2}}(0,-\cos\theta\cos\phi+i\sin\phi,-\cos\theta\sin\phi-i\cos\phi,\sin\theta)\;,

\displaystyle\epsilon^{\mu}(-1)=

\displaystyle\frac{1}{\sqrt{2}}(0,\cos\theta\cos\phi+i\sin\phi,\cos\theta\sin\phi-i\cos\phi,-\sin\theta)\;,

\displaystyle\epsilon^{\mu}(0)=

\displaystyle\frac{1}{\sqrt{m}}(\left|\vec{p}\right|,p^{0}\sin\theta\cos\phi,p^{0}\sin\theta\sin\phi,p^{0}\cos\theta)\;.

(132)

The vectorial spinor
U_{\alpha}(\vec{p},S_{z})

\displaystyle U_{\alpha}\left(\vec{p},S_{z}\right)=\sum_{s_{z},s_{z}^{\prime}}\left\langle s=1,s_{z};s^{\prime}=\frac{1}{2},s_{z}^{\prime}\mid s=1,s^{\prime}=\frac{1}{2};S=\frac{3}{2},S_{z}\right\rangle\varepsilon_{\alpha}\left(\vec{p},s_{z}\right)u\left(\vec{p},s_{z}^{\prime}\right),

\displaystyle S_{z}=\frac{3}{2}:~~~

\displaystyle p_{2},s_{2})\left[g_{0}\frac{m_{2}}{s_{+}}\frac{(m_{2}-m_{1})p_{1}^{\alpha}q^{\mu}}{q^{2}}\right.

\displaystyle+g_{+}\frac{m_{2}}{s_{-}}\frac{(m_{2}+m_{1})p^{\alpha}\big(q^{2}(p_{1}^{\mu}+p_{2}^{\mu})-(m_{1}^{2}-m_{2}^{2})q^{\mu}\big)}{q^{2}s_{-}}

\displaystyle+g_{\perp}\frac{m_{2}}{s_{-}}\left(p^{\alpha}\gamma^{\mu}-\frac{2p^{\alpha}(m_{1}p_{2}^{\mu}+m_{2}p_{1}^{\mu})}{s_{+}}\right)

\displaystyle+\left.g_{\perp^{{}^{\prime}}}\frac{m_{2}}{s_{-}}\left(p^{\alpha}\gamma^{\mu}-\frac{2p_{1}^{\alpha}p_{2}^{\mu}}{m_{2}}+\frac{2p_{1}^{\alpha}(m_{1}p_{2}^{\mu}+m_{2}p^{\mu})}{s_{-}}\right)+\frac{s_{-}g^{\alpha\mu}}{m_{2}}\right]u(p_{1},s_{1})\;.
